  • __device__ 및 __constant__ 변수는 CUDA Runtime API 함수 cudaGetSymbolAddress(), cudaGetSymbolSize(), cudaMemcpyToSymbol(), cudaMemcpyFromSymbol()를 사용하여 호스트에서 접근할 수 있습니다.

  • __constant__ 변수는 디바이스 코드에서 읽기 전용이며, CUDA Runtime API를 사용해 호스트에서만 수정할 수 있습니다.

다음 예는 이러한 API를 사용하는 방법을 보여줍니다:

__device__   float device_var       = 4.0f; // Variable in device memory
__constant__ float constant_mem_var = 4.0f; // Variable in constant memory
                                            // For readability, the following example focuses on a device variable.
int main() {
    float* device_ptr;
    cudaGetSymbolAddress((void**) &device_ptr, device_var);        // Gets address of device_var

    size_t symbol_size;
    cudaGetSymbolSize(&symbol_size, device_var);                   // Retrieves the size of the symbol (4 bytes).

    float host_var;
    cudaMemcpyFromSymbol(&host_var, device_var, sizeof(host_var)); // Copies from device to host.

    host_var = 3.0f;
    cudaMemcpyToSymbol(device_var, &host_var, sizeof(host_var));   // Copies from host to device.
}

5.4.1.2.1. __shared__ 메모리#

__shared__ 메모리 변수는 컴파일 시간에 결정되는 정적 크기를 가질 수도 있고, 커널 실행 시에 결정되는 동적 크기를 가질 수도 있습니다. 런타임에 공유 메모리 크기를 지정하는 방법에 대한 자세한 내용은 커널 구성 섹션을 참조하세요.

공유 메모리 제약 사항:

  • 동적 크기를 갖는 변수는 외부 배열 또는 포인터로 선언해야 합니다.

  • 정적 크기를 갖는 변수는 선언 시 초기화할 수 없습니다.

다음 예는 __shared__ 변수를 선언하고 크기를 지정하는 방법을 보여줍니다:

extern __shared__ char dynamic_smem_pointer[];
// extern __shared__ char* dynamic_smem_pointer; alternative syntax

__global__ void kernel() { // or a __device__ function
    __shared__ int smem_var1[4];                  // static size
    auto smem_var2 = (int*) dynamic_smem_pointer; // dynamic size
}

int main() {
    size_t shared_memory_size = 16;
    kernel<<<1, 1, shared_memory_size>>>();
    cudaDeviceSynchronize();
}

5.4.1.2.2. __managed__ 메모리#

__managed__ 변수에는 다음 제한 사항이 있습니다:

  • __managed__ 변수의 주소는 상수 표현식이 아닙니다.

  • __managed__ 변수는 참조 타입 T&을 가질 수 없습니다.

  • __managed__ 변수의 주소 또는 값은 CUDA 런타임이 유효한 상태가 아닐 수 있는 경우에는 사용하면 안 되며, 여기에는 다음 경우가 포함됩니다:

    • static 또는 thread_local 저장 기간을 갖는 객체의 정적/동적 초기화 또는 소멸에서.

    • exit()가 호출된 이후에 실행되는 코드에서. 예를 들어, __attribute__((destructor))로 표시된 함수.

    • CUDA 런타임이 초기화되지 않았을 수 있을 때 실행되는 코드에서. 예를 들어, __attribute__((constructor))로 표시된 함수.

  • __managed__ 변수는 괄호로 둘러싸지 않은 id-expression 인수로서 decltype() 표현식에 사용할 수 없습니다.

  • __managed__ 변수는 동적으로 할당된 관리 메모리에 대해 지정된 것과 동일한 코히어런스 및 일관성 동작을 가집니다.

  • 지역 변수에 대한 제한 사항도 참고하세요.

다음은 __managed__ 변수의 합법적/불법적 사용 예입니다:

#include <cassert>

__device__ __managed__ int global_var = 10; // OK

int* ptr = &global_var;                     // ERROR: use of a managed variable in static initialization

struct MyStruct1 {
    int field;
    MyStruct1() : field(global_var) {};
};

struct MyStruct2 {
    ~MyStruct2() { global_var = 10; }
};

MyStruct1 temp1; // ERROR: use of managed variable in dynamic initialization

MyStruct2 temp2; // ERROR: use of managed variable in the destructor of
                 //        object with static storage duration

__device__ __managed__ const int const_var = 10;         // ERROR: const-qualified type

__device__ __managed__ int&      reference = global_var; // ERROR: reference type

template <int* Addr>
struct MyStruct3 {};

MyStruct3<&global_var> temp;     // ERROR: address of managed variable is not a constant expression

__global__ void kernel(int* ptr) {
    assert(ptr == &global_var);  // OK
    global_var = 20;             // OK
}

int main() {
    int* ptr = &global_var;      // OK
    kernel<<<1, 1>>>(ptr);
    cudaDeviceSynchronize();
    global_var++;                // OK
    decltype(global_var) var1;   // ERROR: managed variable used as unparenthesized argument to decltype

    decltype((global_var)) var2; // OK
}

5.4.1.3. 인라인 지정자#

다음 지정자는 __host__ 및 __device__ 함수에 대한 인라인화를 제어하는 데 사용할 수 있습니다:

  • __noinline__: nvcc에게 함수를 인라인화하지 않도록 지시합니다.

  • __forceinline__: 단일 번역 단위(translation unit) 내에서 nvcc가 함수를 인라인화하도록 강제합니다.

  • __inline_hint__: 링크 시간 최적화를 사용할 때 번역 단위 간에 공격적인 인라인화를 활성화합니다.

이들 지정자는 상호 배타적입니다.

5.4.1.4. __restrict__ 포인터#

nvcc는 __restrict__ 키워드를 통해 제한 포인터(restricted pointer)를 지원합니다.

포인터 별칭(pointer aliasing)은 두 개 이상의 포인터가 서로 겹치는 메모리 영역을 참조할 때 발생합니다. 이는 코드 재정렬 및 공통 부분식 제거와 같은 최적화를 방해할 수 있습니다.

restrict로 한정된 포인터(restrict-qualified pointer)는 포인터의 수명 동안 해당 포인터가 가리키는 메모리는 오직 그 포인터를 통해서만 접근된다는 프로그래머의 약속입니다. 이를 통해 컴파일러는 더 공격적인 최적화를 수행할 수 있습니다.

  • 디바이스 함수에 접근하는 모든 스레드는 그 함수로부터 읽기만 하거나; 또는

  • 최대 하나의 스레드만 여기에 쓰기를 수행하며, 다른 어떤 스레드도 여기에서 읽지 않습니다.

다음 예는 별칭 문제를 설명하고, 제한 포인터를 사용하면 컴파일러가 명령어 수를 줄이는 데 어떻게 도움이 되는지 보여줍니다:

__device__
void device_function(const float* a, const float* b, float* c) {
    c[0] = a[0] * b[0];
    c[1] = a[0] * b[0];
    c[2] = a[0] * b[0] * a[1];
    c[3] = a[0] * a[1];
    c[4] = a[0] * b[0];
    c[5] = b[0];
    ...
}

포인터가 a, b, 그리고 c 별칭(aliasing)될 수 있으므로, 이를 통한 모든 쓰기 c 의 요소를 수정할 수 a 또는 b. 기능적 정확성을 보장하기 위해, 컴파일러는 로드할 수 없습니다 a[0] 그리고 b[0] 레지스터로 로드하고, 곱한 다음, 결과를 둘 다에 저장할 수 없습니다 c[0] 그리고 c[1]. 이는 결과가 다음과 같은 경우 추상 실행 모델과 달라지기 때문입니다 a[0] 그리고 c[0] 같은 위치에 있었기 때문입니다. 컴파일러는 공통 부분식(common sub-expression)을 활용할 수 없습니다. 마찬가지로, 컴파일러는 다음의 연산 계산을 재정렬할 수 없습니다 c[4] 의 연산 계산과 함께 c[0] 및 c[1] 선행 쓰기가 있기 때문에 c[3] 의 연산에 대한 입력을 변경할 수 있기 때문에 c[4].

a, b, 그리고 c를 제한 포인터로 선언함으로써, 프로그래머는 컴파일러에 해당 포인터들이 별칭(aliasing)되지 않음을 알립니다. 이는 c에 쓰기를 수행해도 a 또는 b의 요소를 절대 덮어쓰지 않는다는 의미입니다. 이에 따라 함수 프로토타입은 다음과 같이 변경됩니다:

__device__
void device_function(const float* __restrict__ a, const float* __restrict__ b, float* __restrict__ c);

컴파일러 최적화기가 효과적으로 동작하려면 모든 포인터 인자가 제한되어야 함에 유의하세요. __restrict__ 키워드를 추가하면, 컴파일러는 추상 실행 모델과 동일한 기능을 유지하면서 임의로 재정렬을 수행하고 공통 부분식 제거(common sub-expression elimination)를 수행할 수 있습니다.

__device__
void device_function(const float* __restrict__ a, const float* __restrict__ b, float* __restrict__ c) {
    float t0 = a[0];
    float t1 = b[0];
    float t2 = t0 * t1;
    float t3 = a[1];
    c[0]     = t2;
    c[1]     = t2;
    c[4]     = t2;
    c[2]     = t2 * t3;
    c[3]     = t0 * t3;
    c[5]     = t1;
    ...
}

Compiler Explorer에서 예제를 확인하세요.

그 결과, 메모리 접근과 연산의 수가 줄어들며, 로드와 공통 부분식을 레지스터에 캐시함으로써 레지스터 압박이 증가하는 것과 균형을 이룹니다.

레지스터 압박은 많은 CUDA 코드에서 중요한 문제이므로, 제한 포인터의 사용은 점유율(occupancy)을 낮춰 성능에 부정적인 영향을 줄 수 있습니다.

__global__ 함수의 const 포인터 중 __restrict__로 표시된 포인터에 대한 접근은, PTX ld.global.nc 또는 __ldg() low-level load and store functions 명령과 유사하게 읽기 전용 캐시 로드로 컴파일됩니다.

__global__
void kernel1(const float* in, float* out) {
    *out = *in; // PTX: ld.global
}

__global__
void kernel2(const float* __restrict__ in, float* out) {
    *out = *in;  // PTX: ld.global.nc
}

5.4.1.5. __grid_constant__ 매개변수#

__global__ 함수 매개변수에 __grid_constant__를 애너테이션하면, 컴파일러가 매개변수의 스레드별 복사본을 생성하는 것을 방지합니다. 대신 그리드의 모든 스레드는 단일 주소를 통해 매개변수에 접근하게 되며, 이는 성능을 개선할 수 있습니다.

__grid_constant__ 매개변수는 다음 속성을 가집니다:

  • 커널의 수명을 가집니다.

  • 단일 커널에만 프라이빗하며, 이는 서브 그리드를 포함해 다른 그리드의 스레드가 해당 객체에 접근할 수 없음을 의미합니다.

  • 커널의 모든 스레드는 동일한 주소를 봅니다.

  • 읽기 전용입니다. __grid_constant__ 객체 또는 mutable 멤버를 포함한 그 하위 객체를 수정하는 것은 정의되지 않은 동작입니다.

요구 사항:

  • __grid_constant__로 애너테이션된 커널 매개변수는 참조가 아닌 타입이면서 const-한정된 타입이어야 합니다.

  • 모든 함수 선언은 어떤 __grid_constant__ 매개변수와도 일관되어야 합니다.

  • 함수 템플릿 특수화는 어떤 __grid_constant__ 매개변수에 대해서도 주 템플릿 선언과 일치해야 합니다.

  • 함수 템플릿 인스턴스화 또한 어떤 __grid_constant__ 매개변수에 대해서도 주 템플릿 선언과 일치해야 합니다.

예:

struct MyStruct {
    int         x;
    mutable int y;
};

__device__ void external_function(const MyStruct&);

__global__ void kernel(const __grid_constant__ MyStruct s) {
    // s.x++; // Compile error: tried to modify read-only memory
    // s.y++; // Undefined Behavior: tried to modify read-only memory

    // Compiler will NOT create a per-thread local copy of "s":
    external_function(s);
}

5.4.2.1. 호스트 컴파일러 타입 확장#

CUDA에서는 호스트 컴파일러가 지원하는 한 비표준 산술 타입의 사용이 허용됩니다. 다음 타입이 지원됩니다:

  • 128비트 정수 타입 __int128.

    • 호스트 컴파일러가 __SIZEOF_INT128__ 매크로를 정의하는 경우 Linux에서 지원됩니다.

  • 128비트 부동소수점 타입 __float128 및 _Float128은 compute capability 10.0 이후의 GPU 디바이스에서 사용할 수 있습니다. __float128 타입의 상수 표현식은 컴파일러에 의해 더 낮은 정밀도의 부동소수점 표현으로 처리될 수 있습니다.

    • 호스트 컴파일러가 __SIZEOF_FLOAT128__ 또는 __FLOAT128__ 매크로를 정의하는 경우 Linux x86에서 지원됩니다.

  • _Complex 타입은 호스트 코드에서만 지원됩니다.

5.4.2.2. 내장 변수#

x, y, z 차원을 따라 그리드와 블록에 대한 커널 구성을 지정하고 가져오는 데 사용되는 값은 dim3 타입입니다. 블록 및 스레드 인덱스를 얻는 데 사용되는 변수는 uint3 타입입니다. dim3 및 uint3는 모두 x, y, z라는 이름의 세 개의 부호 없는 값으로 구성된 트리비얼 구조체입니다. C++11 이후에서는 dim3의 모든 구성 요소의 기본값이 1입니다.

디바이스 전용 내장 변수:

  • dim3 gridDim: x, y, z 차원을 따라(즉, 스레드 블록의 개수) 그리드의 차원을 포함합니다.

  • dim3 blockDim: x, y, z 차원을 따라(즉, 스레드의 개수) 스레드 블록의 차원을 포함합니다.

  • uint3 blockIdx: x, y, z 차원을 따라 그리드 내에서의 블록 인덱스를 포함합니다.

  • uint3 threadIdx: x, y, z 차원을 따라 블록 내에서의 스레드 인덱스를 포함합니다.

  • int warpSize : 워프의 스레드 수로 정의되는 런타임 값으로, 일반적으로 32입니다. 워프의 정의는 Warps and SIMT도 참고하세요.

5.4.3. 커널 구성#

__global__ 함수에 대한 모든 호출은 해당 호출의 실행 구성을 지정해야 합니다. 이 실행 구성은 디바이스에서 함수를 실행하는 데 사용될 그리드와 블록의 차원을 정의하며, 관련 스트림도 정의합니다.

실행 구성은 함수 이름과 괄호로 둘러싸인 인자 목록 사이에 <<<grid_dim, block_dim, dynamic_smem_bytes, stream>>> 형태의 표현식을 삽입하여 지정하며, 여기서:

  • grid_dim는 dim3 타입이며, 그리드의 차원과 크기를 지정하여 grid_dim.x * grid_dim.y * grid_dim.z가 실행(launch)되는 블록 수와 같아지도록 합니다;

  • block_dim는 dim3 타입이며, 각 블록의 차원과 크기를 지정하여 block_dim.x * block_dim.y * block_dim.z가 블록당 스레드 수와 같아지도록 합니다;

  • dynamic_smem_bytes는 기본값이 0인 선택적 size_t 인자입니다. 정적으로 할당된 메모리에 더해, 이 호출을 위해 블록당 동적으로 할당되는 공유 메모리의 바이트 수를 지정합니다. 이 메모리는 extern __shared__ 배열에서 사용됩니다( __shared__ Memory 참조).

  • stream는 cudaStream_t (포인터) 타입이며, 관련 스트림을 지정합니다. stream는 기본값이 NULL인 선택적 인자입니다.

다음 예는 커널 함수 선언과 호출을 보여줍니다:

__global__ void kernel(float* parameter);

kernel<<<grid_dim, block_dim, dynamic_smem_bytes>>>(parameter);

실행 구성의 인자는 실제 함수의 인자보다 먼저 평가됩니다.

grid_dim 또는 block_dim가 컴퓨트 기능에 지정된 바와 같이 해당 디바이스에 허용되는 최대 크기를 초과하거나, 정적으로 할당된 메모리를 고려한 뒤의 사용 가능한 공유 메모리보다 dynamic_smem_bytes가 더 크면 함수 호출은 실패합니다.

5.4.3.1. 스레드 블록 클러스터#

컴퓨트 기능 9.0 이상에서는 사용자가 컴파일 시간(thread block) 스레드 블록 클러스터 차원을 지정할 수 있어, 커널이 CUDA의 cluster hierarchy를 사용할 수 있습니다. 컴파일 시간 클러스터 차원은 다음 구문으로 __cluster_dims__ 속성을 사용해 지정할 수 있습니다: __cluster_dims__([x, [y, [z]]]). 아래 예는 X 차원에서는 컴파일 시간 클러스터 크기가 2이고, Y 및 Z 차원에서는 1인 경우를 보여줍니다.

__global__ void __cluster_dims__(2, 1, 1) kernel(float* parameter);

__cluster_dims__()의 기본 형태는 커널이 그리드 클러스터로 실행되도록 지정합니다. 클러스터 차원을 지정하지 않은 경우, 사용자는 실행 시점에 이를 지정할 수 있습니다. 실행 시점에 차원을 지정하지 않으면 실행 시점 오류가 발생합니다.

스레드 블록 클러스터의 차원은 런타임에도 지정할 수 있으며, 클러스터가 있는 커널은 cudaLaunchKernelEx API를 사용해 실행할 수 있습니다. 이 API는 cudaLaunchConfig_t 타입의 구성 인자, 커널 함수 포인터, 그리고 커널 인자를 받습니다. 아래 예는 런타임 커널 구성을 보여줍니다.

__global__ void kernel(float parameter1, int parameter2) {}

int main() {
    cudaLaunchConfig_t config = {0};
    // The grid dimension is not affected by cluster launch, and is still enumerated
    // using the number of blocks.
    // The grid dimension should be a multiple of cluster size.
    config.gridDim          = dim3{4};  // 4 blocks
    config.blockDim         = dim3{32}; // 32 threads per block
    config.dynamicSmemBytes = 1024;     // 1 KB

    cudaLaunchAttribute attribute[1];
    attribute[0].id               = cudaLaunchAttributeClusterDimension;
    attribute[0].val.clusterDim.x = 2; // Cluster size in X-dimension
    attribute[0].val.clusterDim.y = 1;
    attribute[0].val.clusterDim.z = 1;
    config.attrs    = attribute;
    config.numAttrs = 1;

    float parameter1 = 3.0f;
    int   parameter2 = 4;
    cudaLaunchKernelEx(&config, kernel, parameter1, parameter2);
}

Compiler Explorer에서 예제를 확인하세요.

5.4.3.2. 런치 바운드(Launch Bounds)#

Kernel Launch and Occupancy 섹션에서 논의했듯이, 레지스터를 더 적게 사용하면 더 많은 스레드와 스레드 블록이 멀티프로세서에 상주할 수 있어 성능이 향상됩니다.

따라서 컴파일러는 register spilling과 명령어 수를 최소화하면서 레지스터 사용량을 최소화하기 위해 휴리스틱을 사용합니다. 애플리케이션은 __launch_bounds__() 한정자를 사용해 __global__ 함수의 정의에서 지정되는 런치 바운드 형태로 컴파일러에 추가 정보를 제공함으로써, 선택적으로 이러한 휴리스틱을 보조할 수 있습니다:

__global__ void
__launch_bounds__(maxThreadsPerBlock, minBlocksPerMultiprocessor, maxBlocksPerCluster)
MyKernel(...) {
    ...
}

  • maxThreadsPerBlock는 애플리케이션이 MyKernel()를 런치할 때 사용할 수 있는 블록당 최대 스레드 수를 지정합니다. 이는 .maxntid PTX 지시문으로 컴파일됩니다.

  • minBlocksPerMultiprocessor는 선택 사항이며, 멀티프로세서당 상주 블록의 원하는 최소 개수를 지정합니다. 이는 .minnctapersm PTX 지시문으로 컴파일됩니다.

  • maxBlocksPerCluster는 선택 사항이며, 애플리케이션이 MyKernel()를 런치할 때 사용할 수 있는 클러스터당 스레드 블록의 원하는 최대 개수를 지정합니다. 이는 .maxclusterrank PTX 지시문으로 컴파일됩니다.

런치 바운드(launch bounds)가 지정되면, 컴파일러는 먼저 상한을 도출합니다, L, 커널이 사용해야 하는 레지스터 개수에 대한 상한을. 이는 다음을 보장합니다 minBlocksPerMultiprocessor 블록(또는, 만약 minBlocksPerMultiprocessor 지정되지 않은 경우)의 maxThreadsPerBlock 스레드가 멀티프로세서에 상주할 수 있습니다. 다음을 참조하세요 점유율(occupancy) 섹션에서 커널이 사용하는 레지스터 수와 블록당 할당되는 레지스터 수 간의 관계를 확인하세요. 그런 다음 컴파일러는 다음과 같이 레지스터 사용을 최적화합니다:

  • 초기 레지스터 사용량이 L를 초과하면, 컴파일러는 L 이하가 될 때까지 이를 줄입니다. 이는 보통 로컬 메모리 사용량 증가 및/또는 더 많은 명령 수로 이어집니다.

  • 초기 레지스터 사용량이 L보다 낮으면

    • 만약 maxThreadsPerBlock 지정되었지만 minBlocksPerMultiprocessor 지정되지 않으면, 컴파일러는 사용합니다 maxThreadsPerBlock … 간 전환을 위한 레지스터 사용 임계값을 결정하기 위해 n 그리고 n + 1 상주 블록. 이는 레지스터를 하나 덜 사용하면 추가 상주 블록을 위한 공간이 생길 때 발생합니다. 그런 다음, 컴파일러는 런치 바운드(launch bounds)가 지정되지 않은 경우와 유사한 휴리스틱을 적용합니다.

    • 둘 다 minBlocksPerMultiprocessor 그리고 maxThreadsPerBlock 가 지정되면, 컴파일러는 레지스터 사용량을 다음까지 늘릴 수 있습니다 L 명령 수를 줄이고 단일 스레드 명령의 지연 시간을 더 잘 숨기기 위해.

커널은 다음 조건으로 실행되면 런치에 실패합니다:

  • 블록당 스레드 수가 해당 런치 바운드(launch bound) maxThreadsPerBlock보다 많은 경우.

  • 클러스터당 스레드 블록 수가 해당 런치 바운드(launch bound) maxBlocksPerCluster보다 많은 경우.

CUDA 커널에 스레드당 필요로 하는 리소스는 최대 블록 크기를 바람직하지 않은 방식으로 제한할 수 있습니다. 향후 하드웨어 및 툴킷과의 전방 호환성을 유지하고, 최소한 하나의 스레드 블록이 스트리밍 멀티프로세서에서 실행될 수 있도록 보장하기 위해, 개발자는 커널이 런치될 가장 큰 블록 크기를 지정하는 단일 인자 __launch_bounds__(maxThreadsPerBlock)를 포함해야 합니다. 이렇게 하지 않으면 “too many resources requested for launch” 오류가 발생할 수 있습니다. __launch_bounds__(maxThreadsPerBlock,minBlocksPerMultiprocessor)의 두 인자 버전을 제공하면 경우에 따라 성능이 향상될 수 있습니다. minBlocksPerMultiprocessor의 최적 값은 각 커널에 대한 상세 분석을 통해 결정해야 합니다.

커널에 대한 최적의 런치 바운드(launch bounds)는 일반적으로 주요 아키텍처 개정 버전마다 서로 다릅니다. 다음 코드 샘플은 디바이스 코드에서 __CUDA_ARCH__ macro로 이를 어떻게 관리하는지 보여줍니다.

#define THREADS_PER_BLOCK  256

#if __CUDA_ARCH__ >= 900
    #define MY_KERNEL_MAX_THREADS  (2 * THREADS_PER_BLOCK)
    #define MY_KERNEL_MIN_BLOCKS   3
#else
    #define MY_KERNEL_MAX_THREADS  THREADS_PER_BLOCK
    #define MY_KERNEL_MIN_BLOCKS   2
#endif

__global__ void
__launch_bounds__(MY_KERNEL_MAX_THREADS, MY_KERNEL_MIN_BLOCKS)
MyKernel(...) {
    ...
}

때 MyKernel 블록당 최대 스레드 수로 호출되는 경우로, 이는 첫 번째 매개변수로 지정됩니다 __launch_bounds__(), 이를 사용하고 싶은 유혹이 생깁니다 MY_KERNEL_MAX_THREADS 실행 구성에서 블록당 스레드 수로서:

// Host code
MyKernel<<<blocksPerGrid, MY_KERNEL_MAX_THREADS>>>(...);

하지만 이는 동작하지 않을 것입니다. 왜냐하면 __CUDA_ARCH__ 에서 언급했듯이 호스트 코드에서는 정의되지 않습니다 실행 공간 지정자 섹션. 따라서, MyKernel 블록당 256개의 스레드로 런치됩니다. 대신 블록당 스레드 수는 다음과 같이 결정되어야 합니다:

  • 예를 들어 __CUDA_ARCH__에 의존하지 않는 매크로 또는 상수를 사용해 컴파일 시간에 결정하거나

    // Host code
MyKernel<<<blocksPerGrid, THREADS_PER_BLOCK>>>(...);

  • 또는 컴퓨트 기능(compute capability)에 기반하여 런타임에

    // Host code
cudaGetDeviceProperties(&deviceProp, device);
int threadsPerBlock = (deviceProp.major >= 9) ? 2 * THREADS_PER_BLOCK : THREADS_PER_BLOCK;
MyKernel<<<blocksPerGrid, threadsPerBlock>>>(...);

--resource-usage 컴파일러 옵션은 레지스터 사용량을 보고합니다. CUDA 프로파일러는 점유율(occupancy)을 보고하며, 이는 상주 블록 수를 도출하는 데 사용할 수 있습니다.

5.4.3.3. 스레드당 최대 레지스터 수#

저수준 성능 튜닝을 활성화하기 위해, CUDA C++는 백엔드 최적화 컴파일러에 성능 튜닝 정보를 전달하는 __maxnreg__() 함수 한정자(qualifier)를 제공합니다. __maxnreg__() 한정자는 스레드 블록의 단일 스레드에 할당될 수 있는 최대 레지스터 수를 지정합니다. __global__ 함수의 정의에서:

__global__ void
__maxnreg__(maxNumberRegistersPerThread)
MyKernel(...) {
    ...
}

maxNumberRegistersPerThread 변수는 커널 MyKernel()의 스레드 블록에서 단일 스레드에 할당될 최대 레지스터 수를 지정하며; 이는 .maxnreg PTX 지시문(directive)으로 컴파일됩니다.

__launch_bounds__() 및 __maxnreg__() 한정자는 동일한 커널에 함께 적용할 수 없습니다.

--maxrregcount <N> 컴파일러 옵션은 파일 내 모든 __global__ 함수에 대해 레지스터 사용량을 제어하는 데 사용할 수 있습니다. 이 옵션은 __maxnreg__ 한정자가 있는 커널 함수에 대해서는 무시됩니다.

5.4.4. 동기화 프리미티브#

5.4.4.1. 스레드 블록 동기화 함수#

void __syncthreads();
int  __syncthreads_count(int predicate);
int  __syncthreads_and(int predicate);
int  __syncthreads_or(int predicate);

이 내장 함수(intrinsic)는 동일한 블록 내 스레드 간 통신을 조정합니다. 블록의 스레드가 공유 메모리 또는 전역 메모리에서 동일한 주소에 접근할 때, write-after-read, read-after-write 또는 write-after-write 해저드(hazard)가 발생할 수 있습니다. 이러한 해저드는 이러한 접근 사이에서 스레드를 동기화함으로써 피할 수 있습니다.

이 내장 함수(intrinsic)는 다음과 같은 의미론(semantics)을 가집니다:

  • __syncthreads*() 스레드 블록에서 종료되지 않은 모든 스레드가 프로그램에서 동일한 __syncthreads*() intrinsic 호출에 동시에 도달하거나 종료할 때까지 기다립니다.

  • __syncthreads*() 참여하는 스레드들 간의 메모리 순서를 제공합니다: __syncthreads*() intrinsic에 대한 호출은 어떤 참여 스레드가 대기에서 언블록되거나 종료하기 전에 강하게 happens before 관계를 가집니다( C++ specification [intro.races] 참조).

다음 예는 __syncthreads()를 사용하여 스레드 블록 내 스레드를 동기화하고 스레드 간에 공유되는 배열의 요소를 안전하게 합산하는 방법을 보여줍니다:

// assuming blockDim.x is 128
__global__ void example_syncthreads(int* input_data, int* output_data) {
    __shared__ int shared_data[128];
    // Every thread writes to a distinct element of 'shared_data':
    shared_data[threadIdx.x] = input_data[threadIdx.x];

    // All threads synchronize, guaranteeing all writes to 'shared_data' are ordered 
    // before any thread is unblocked from '__syncthreads()':
    __syncthreads();

    // A single thread safely reads 'shared_data':
    if (threadIdx.x == 0) {
        int sum = 0;
        for (int i = 0; i < blockDim.x; ++i) {
            sum += shared_data[i];
        }
        output_data[blockIdx.x] = sum;
    }
}

__syncthreads*() intrinsic는 조건부 코드에서 허용되지만, 조건이 전체 스레드 블록에 걸쳐 균일하게 평가되는 경우에만 허용됩니다. 그렇지 않으면 실행이 멈추거나 의도하지 않은 부작용이 발생할 수 있습니다.

다음 예는 유효한 동작을 보여줍니다:

// assuming blockDim.x is 128
__global__ void syncthreads_valid_behavior(int* input_data, int* output_data) {
    __shared__ int shared_data[128];
    shared_data[threadIdx.x] = input_data[threadIdx.x];
    if (blockIdx.x > 0) { // CORRECT, uniform condition across all block threads
        __syncthreads();
        output_data[threadIdx.x] = shared_data[128 - threadIdx.x];
    }
}

반면 다음 예들은 커널 멈춤과 같은 잘못된 동작 또는 정의되지 않은 동작을 보입니다:

// assuming blockDim.x is 128
__global__ void syncthreads_invalid_behavior1(int* input_data, int* output_data) {
    __shared__ int shared_data[256];
    shared_data[threadIdx.x] = input_data[threadIdx.x];
    if (threadIdx.x > 0) { // WRONG, non-uniform condition
        __syncthreads();   // Undefined Behavior
        output_data[threadIdx.x] = shared_data[128 - threadIdx.x];
    }
}

// assuming blockDim.x is 128
__global__ void syncthreads_invalid_behavior2(int* input_data, int* output_data) {
    __shared__ int shared_data[256];
    shared_data[threadIdx.x] = input_data[threadIdx.x];
    for (int i = 0; i < blockDim.x; ++i) {
        if (i == threadIdx.x) { // WRONG, non-uniform condition
            __syncthreads();    // Undefined Behavior
        }
    }
    output_data[threadIdx.x] = shared_data[128 - threadIdx.x];
}

__syncthreads() predicate를 사용하는 변형:

int __syncthreads_count(int predicate);

이는 블록에서 종료되지 않은 모든 스레드에 대해 predicate를 평가하고, predicate가 0이 아닌 값으로 평가되는 스레드의 수를 반환한다는 점을 제외하면 __syncthreads()와 동일합니다.

int __syncthreads_and(int predicate);

는 블록에서 종료되지 않은 모든 스레드에 대해 predicate를 평가한다는 점을 제외하면 __syncthreads()와 동일합니다. 이는 그들 모두에 대해 predicate가 0이 아닌 값으로 평가되는 경우에만, 그리고 그 경우에만 0이 아닌 값을 반환합니다.

int __syncthreads_or(int predicate);

는 블록에서 종료되지 않은 모든 스레드에 대해 predicate를 평가한다는 점을 제외하면 __syncthreads()와 동일합니다. 이는 그들 중 하나 이상에 대해 predicate가 0이 아닌 값으로 평가되는 경우에만, 그리고 그 경우에만 0이 아닌 값을 반환합니다.

5.4.4.2. 워프 동기화 함수#

void __syncwarp(unsigned mask = 0xFFFFFFFF);

내장 함수(intrinsic function) __syncwarp()는 동일한 워프 내 스레드 간 통신을 조정합니다. 워프 내 일부 스레드가 공유 메모리 또는 전역 메모리에서 동일한 주소에 접근할 때, 잠재적으로 read-after-write, write-after-read 또는 write-after-write 해저드(hazard)가 발생할 수 있습니다. 이러한 데이터 해저드는 이러한 접근 사이에서 스레드를 동기화함으로써 피할 수 있습니다.

__syncwarp(mask)를 호출하면 mask에 이름이 지정된 워프 내 참여 스레드들 간의 메모리 순서를 제공합니다: __syncwarp(mask)에 대한 호출은( C++ specification [intro.races] 참조) mask에 이름이 지정된 어떤 워프 스레드가 대기에서 언블록되거나 종료하기 전에 강하게 happens before 관계를 가집니다.

이 함수는 Warp __sync Intrinsic Constraints의 제약을 받습니다.

다음 예는 워프 내 스레드를 동기화하여 공유 메모리 배열에 안전하게 접근하기 위해 __syncwarp()를 사용하는 방법을 보여줍니다:

__global__ void example_syncwarp(int* input_data, int* output_data) {
    if (threadIdx.x < warpSize) {
        __shared__ int shared_data[warpSize];
        shared_data[threadIdx.x] = input_data[threadIdx.x];

        __syncwarp(); // equivalent to __syncwarp(0xFFFFFFFF)
        if (threadIdx.x == 0)
            output_data[0] = shared_data[1];
    }
}

5.4.4.3. 메모리 펜스 함수#

CUDA 프로그래밍 모델은 약한 순서(weakly ordered) 메모리 모델을 가정합니다. 즉, CUDA 스레드가 공유 메모리, 전역 메모리, 페이지-락 호스트 메모리 또는 피어 디바이스의 메모리에 데이터를 쓰는 순서는, 다른 CUDA 또는 호스트 스레드가 쓰여지는 데이터를 관찰하는 순서와 반드시 같지는 않습니다. 메모리 펜스 또는 동기화 없이 동일한 메모리 위치에서 읽기 또는 쓰기를 수행하면 정의되지 않은 동작(undefined behavior)이 발생합니다.

다음 예에서 스레드 1은 writeXY()를 실행하고, 스레드 2는 readXY()를 실행합니다.

__device__ int X = 1, Y = 2;

__device__ void writeXY() {
    X = 10;
    Y = 20;
}

__device__ void readXY() {
    int B = Y;
    int A = X;
}

두 스레드는 동일한 메모리 위치인 X 및 Y에 대해 동시에 읽기와 쓰기를 수행합니다. 어떤 데이터 레이스든 정의되지 않은 동작(undefined behavior)을 초래하며 정의된 의미론(semantics)이 없습니다. 따라서 A 및 B에 대한 결과 값은 무엇이든 될 수 있습니다.

메모리 펜스 및 동기화 함수는 메모리 접근에 대해 순차적 일관성(sequentially consistent) 순서를 강제합니다. 이러한 함수는 순서가 강제되는 스레드 스코프(thread scope)에서 차이가 있지만, 공유 메모리, 전역 메모리, 페이지-락 호스트 메모리 및 피어 디바이스의 메모리를 포함하여 접근되는 메모리 공간과는 무관합니다.

힌트

사용할 것을 권장합니다 cuda::atomic_thread_fence 제공되는 libcu++ 안전성과 이식성 이유로 가능할 때마다.

__device__ void writeXY() {
    X = 10;
    cuda::atomic_thread_fence(cuda::memory_order_seq_cst, cuda::thread_scope_device);
    Y = 20;
}

__device__ void readXY() {
    int B = Y;
    cuda::atomic_thread_fence(cuda::memory_order_seq_cst, cuda::thread_scope_device);
    int A = X;
}

__device__ int X = 1, Y = 2;

__device__ void writeXY() {
    X = 10;
    __threadfence();
    Y = 20;
}

__device__ void readXY() {
    int B = Y;
    __threadfence();
    int A = X;
}

이 코드에서는 다음과 같은 결과를 관찰할 수 있습니다:

  • A는 1이고 B는 2이며, 즉 readXY()가 writeXY()보다 먼저 실행됩니다.

  • A는 10이고 B는 20이며, 즉 writeXY()가 readXY()보다 먼저 실행됩니다.

  • A는 10이고 B는 2입니다.

  • A가 1이고 B가 20인 경우는, 메모리 펜스가 X에 대한 쓰기가 Y에 대한 쓰기보다 먼저 가시적임을 보장하므로 불가능합니다.

스레드 1과 2가 동일한 블록에 속한다면 블록 수준 펜스만 사용해도 충분합니다. 스레드 1과 2가 동일한 블록에 속하지 않는다면, 같은 디바이스의 CUDA 스레드인 경우에는 디바이스 수준 펜스를 사용해야 하고, 서로 다른 두 디바이스의 CUDA 스레드인 경우에는 시스템 수준 펜스를 사용해야 합니다.

일반적인 사용 사례는 다음 코드 샘플로 설명할 수 있으며, 여기서 스레드는 다른 스레드가 생성한 데이터를 소비합니다. 이 커널은 N개의 숫자로 이루어진 배열의 합을 단일 호출로 계산합니다.

  • 각 블록은 먼저 배열의 부분 집합을 합산하고 그 결과를 전역 메모리에 저장합니다.

  • 모든 블록이 완료되면, 마지막 블록은 전역 메모리에서 이러한 각 부분 합을 읽어 함께 더해 최종 결과를 얻습니다.

  • 어떤 블록이 마지막에 완료되었는지 결정하기 위해, 각 블록은 부분 합의 계산 및 저장 완료를 알리기 위해 카운터를 원자적으로 증가시킵니다(자세한 내용은 Atomic Functions 섹션 참조). 마지막 블록은 gridDim.x - 1와 동일한 카운터 값을 받습니다.

부분 합을 저장하는 것과 카운터를 증가시키는 것 사이에 펜스가 없으면, 부분 합이 저장되기 전에 카운터가 증가할 수 있습니다. 이로 인해 카운터가 gridDim.x - 1에 도달하고, 마지막 블록이 메모리에서 업데이트되기 전에 부분 합을 읽기 시작할 수 있습니다.

참고

메모리 펜스는 메모리 연산이 실행되는 순서에만 영향을 미치며, 이러한 연산이 다른 스레드에 가시적임을 보장하지는 않습니다.

아래 코드 샘플에서는 result 변수에 대한 메모리 연산의 가시성을, 이를 volatile로 선언함으로써 보장합니다. 자세한 내용은 volatile-qualified variables 섹션을 참조하십시오.

#include <cuda/atomic>

__device__ int count = 0;

__global__ void sum(const float*    array,
                    int             N,
                    volatile float* result) {
    __shared__ bool isLastBlockDone;
    // Each block sums a subset of the input array.
    float partialSum = calculatePartialSum(array, N);

    if (threadIdx.x == 0) {
        // Thread 0 of each block stores the partial sum to global memory.
        // The compiler will use a store operation that bypasses the L1 cache
        // since the "result" variable is declared as volatile.
        // This ensures that the threads of the last block will read the correct
        // partial sums computed by all other blocks.
        result[blockIdx.x] = partialSum;

        // Thread 0 makes sure that the increment of the "count" variable is
        // only performed after the partial sum has been written to global memory.
        cuda::atomic_thread_fence(cuda::memory_order_seq_cst, cuda::thread_scope_device);

        // Thread 0 signals that it is done.
        int count_old = atomicInc(&count, gridDim.x);

        // Thread 0 determines if its block is the last block to be done.
        isLastBlockDone = (count_old == (gridDim.x - 1));
    }
    // Synchronize to make sure that each thread reads the correct value of
    // isLastBlockDone.
    __syncthreads();

    if (isLastBlockDone) {
        // The last block sums the partial sums stored in result[0 .. gridDim.x-1]
        float totalSum = calculateTotalSum(result);

        if (threadIdx.x == 0) {
            // Thread 0 of last block stores the total sum to global memory and
            // resets the count variable, so that the next kernel call works
            // properly.
            result[0] = totalSum;
            count     = 0;
        }
    }
}

5.4.5. 원자 함수#

원자 함수는 공유 데이터에 대해 읽기-수정-쓰기(read-modify-write) 연산을 수행하여, 단일 단계로 실행되는 것처럼 보이게 합니다. 원자성(atomicity)은 각 연산이 완전히 완료되거나 아예 수행되지 않도록 보장하여, 참여하는 모든 스레드에 데이터에 대한 일관된 뷰를 제공합니다.

CUDA는 네 가지 방식으로 원자 함수를 제공합니다:

확장 CUDA C++ 원자 함수, cuda::atomic 및 cuda::atomic_ref.

  • 호스트 코드와 디바이스 코드 모두에서 사용할 수 있습니다.

  • C++ standard atomic operations 의미론(semantics)을 따릅니다.

  • 원자 연산의 thread scope를 지정할 수 있습니다.

표준 C++ 원자 함수, cuda::std::atomic 및 cuda::std::atomic_ref.

  • 호스트 코드와 디바이스 코드 모두에서 사용할 수 있습니다.

  • C++ standard atomic operations 의미론(semantics)을 따릅니다.

  • 원자 연산의 thread scope를 지정할 수 없습니다.

컴파일러 built-in atomic functions, __nv_atomic_<op>().

  • CUDA 12.8부터 사용할 수 있습니다.

  • 디바이스 코드에서만 사용할 수 있습니다.

  • C++ standard atomic memory order 의미론(semantics)을 따릅니다.

  • 원자 연산의 thread scope를 지정할 수 있습니다.

  • C++ standard atomic operations과 동일한 메모리 순서 지정 의미론(semantics)을 가집니다.

  • 128비트 데이터 타입을 제외하고, cuda::std::atomic 및 cuda::std::atomic_ref에서 허용되는 데이터 타입의 하위 집합을 지원합니다.

Legacy atomic functions, atomic<Op>().

  • 디바이스 코드에서만 허용됩니다.

  • memory_order_relaxed C++ atomic memory semantics만 지원합니다.

  • 원자 연산의 thread scope를 함수 이름의 일부로 지정할 수 있습니다.

  • built-in atomic functions와 달리, 레거시 원자 함수는 원자성만 보장하며 동기화 지점(펜스)을 도입하지 않습니다.

  • built-in atomic functions에서 허용되는 데이터 타입의 하위 집합을 지원합니다. 원자 add 연산은 추가 데이터 타입을 지원합니다.

힌트

사용하는 확장 CUDA C++ 원자 함수 제공하는 libcu++ 효율성, 안전성 및 이식성을 위해 권장됩니다.

5.4.5.1. 레거시 원자 함수#

레거시 원자 함수는 전역 또는 공유 메모리에 저장된 32, 64 또는 128비트 워드에 대해 원자적 read-modify-write 연산을 수행합니다. 예를 들어, atomicAdd() 함수는 전역 또는 공유 메모리의 특정 주소에 있는 워드를 읽고, 여기에 숫자를 더한 다음, 그 결과를 동일한 주소에 다시 씁니다.

  • 원자 함수는 디바이스 함수에서만 사용할 수 있습니다.

  • __half2, __nv_bfloat162, float2, float4와 같은 벡터 타입의 경우, read-modify-write 연산은 벡터의 각 요소에서 수행됩니다. 전체 벡터가 단일 접근에서 원자적임이 보장되지는 않습니다.

이 섹션에서 설명하는 원자 함수는 메모리 순서 지정 의 cuda::std::memory_order_relaxed 그리고 특정한 경우에만 원자적입니다 thread scope:

  • 접미사가 없는 Atomic API, 예를 들어 atomicAdd, scope 수준에서 원자적입니다 cuda::thread_scope_device.

  • 다음과 함께하는 Atomic API _block 접미사. 예를 들어, atomicAdd_block, scope에서 원자적입니다 cuda::thread_scope_block.

  • 다음과 함께하는 Atomic API _system 접미사. 예를 들어, atomicAdd_system, scope에서 원자적입니다 cuda::thread_scope_system 특정 조건을 충족하는 경우 조건.

다음 예는 CPU와 GPU가 주소에 있는 정수 값을 원자적으로 업데이트하는 것을 보여줍니다 addr:

#include <cuda_runtime.h>

__global__ void atomicAdd_kernel(int* addr) {
    atomicAdd_system(addr, 10);
}

void test_atomicAdd(int device_id) {
    int* addr;
    cudaMallocManaged(&addr, 4);
    *addr = 0;

    cudaDeviceProp deviceProp;
    cudaGetDeviceProperties(&deviceProp, device_id);
    if (deviceProp.concurrentManagedAccess != 1) {
        return; // the device does not coherently access managed memory concurrently with the CPU
    }

    atomicAdd_kernel<<<...>>>(addr);
    __sync_fetch_and_add(addr, 10);  // CPU atomic operation
}

어떤 원자 연산도 atomicCAS() (Compare and Swap)를 기반으로 구현할 수 있음을 유의하십시오. 예를 들어, 단정밀도 부동소수점 숫자에 대한 atomicAdd()는 다음과 같이 구현할 수 있습니다:

#include <cuda/memory>
#include <cuda/std/bit>

__device__ float customAtomicAdd(float* d_ptr, float value) {
    volatile unsigned* d_ptr_unsigned = reinterpret_cast<unsigned*>(d_ptr);
    unsigned  old_value      = *d_ptr_unsigned;
    unsigned  assumed;
    do {
        assumed                          = old_value;
        float    assumed_float           = cuda::std::bit_cast<float>(assumed);
        float    expected_value          = assumed_float + value;
        unsigned expected_value_unsigned = cuda::std::bit_cast<unsigned>(expected_value);
        old_value                        = atomicCAS(d_ptr_unsigned, assumed, expected_value_unsigned);
    // Note: uses integer comparison to avoid hang in case of NaN (since NaN != NaN)
    } while (assumed != old_value);
    return cuda::std::bit_cast<float>(old_value);
}

  3. 동일한 주소의 메모리에 결과를 다시 저장합니다.

이 함수는 다음을 반환합니다 old 값.

atomicCAS() 다음 데이터 타입을 지원합니다:

  • int, unsigned, unsigned long long, unsigned short.

C++ 템플릿 함수 atomicCAS()는 다음 요구 사항을 충족하는 128비트 타입을 지원합니다:

  • 컴퓨트 기능 9.x 이상.

  • T는 16바이트에 정렬되어야 하며, 즉 alignof(T) >= 16.

  • T는 trivially copyable이어야 하며, 즉 std::is_trivially_copyable_v<T>.

  • C++03 및 이전: T는 trivially constructible이어야 하며, 즉 std::is_default_constructible_v<T>.

5.4.5.2. 내장 원자 함수#

CUDA 12.8 및 이후 버전은 C++ 표준 원자 연산 및 CUDA thread scopes와 동일한 메모리 순서 지정 시맨틱을 따르는 원자 연산을 위한 CUDA 컴파일러 내장 함수를 지원합니다. 이 함수들은 thread scope에 대한 추가 인자를 포함하는 GNU의 원자 내장 함수 시그니처를 따릅니다.

nvcc는 내장 원자 함수가 지원될 때 매크로 __CUDACC_DEVICE_ATOMIC_BUILTINS__를 정의합니다.

아래에는 메모리 순서와 thread scopes에 대한 원시 열거자를 나열했으며, 이는 내장 원자 함수의 order 및 scope 인자로 사용됩니다:

// atomic memory orders
enum {
   __NV_ATOMIC_RELAXED,
   __NV_ATOMIC_CONSUME,
   __NV_ATOMIC_ACQUIRE,
   __NV_ATOMIC_RELEASE,
   __NV_ATOMIC_ACQ_REL,
   __NV_ATOMIC_SEQ_CST
};

// thread scopes
enum {
   __NV_THREAD_SCOPE_THREAD,
   __NV_THREAD_SCOPE_BLOCK,
   __NV_THREAD_SCOPE_CLUSTER,
   __NV_THREAD_SCOPE_DEVICE,
   __NV_THREAD_SCOPE_SYSTEM
};

  • 메모리 순서는 C++ 표준 원자 연산의 메모리 순서에 대응합니다.

  • thread scope는 cuda::thread_scope 정의를 따릅니다.

  • __NV_ATOMIC_CONSUME 메모리 순서는 현재 더 강한 __NV_ATOMIC_ACQUIRE 메모리 순서를 사용하여 구현되어 있습니다.

  • __NV_THREAD_SCOPE_THREAD thread scope는 현재 더 넓은 __NV_THREAD_SCOPE_BLOCK thread scope를 사용하여 구현되어 있습니다.

예:

__device__ T __nv_atomic_load_n(T*  pointer,
                                int memory_order,
                                int thread_scope = __NV_THREAD_SCOPE_SYSTEM);

원자 내장 함수에는 다음과 같은 제한이 있습니다:

  • 디바이스 함수에서만 사용할 수 있습니다.

  • 로컬 메모리에서 동작할 수 없습니다.

  • 이 함수들의 주소는 취할 수 없습니다.

  • order 및 scope 인자는 정수 리터럴이어야 하며, 변수일 수 없습니다.

  • thread scope __NV_THREAD_SCOPE_CLUSTER는 아키텍처 sm_90 이상에서 지원됩니다.

지원되지 않는 경우의 예:

 // Not permitted in a host function
 __host__ void bar() {
     unsigned u1 = 1, u2 = 2;
     __nv_atomic_load(&u1, &u2, __NV_ATOMIC_RELAXED, __NV_THREAD_SCOPE_SYSTEM);
 }

 // Not permitted to be applied to local memory
__device__ void foo() {
   unsigned a = 1, b;
   __nv_atomic_load(&a, &b, __NV_ATOMIC_RELAXED, __NV_THREAD_SCOPE_SYSTEM);
}

 // Not permitted as a template default argument.
 // The function address cannot be taken.
 template<void *F = __nv_atomic_load_n>
 class X {
     void *f = F; // The function address cannot be taken.
 };

 // Not permitted to be called in a constructor initialization list.
 class Y {
     int a;
 public:
     __device__ Y(int *b): a(__nv_atomic_load_n(b, __NV_ATOMIC_RELAXED)) {}
 };

다음 섹션에서는 워프 내의 스레드들이 서로 통신하고 계산을 수행할 수 있게 하는 워프 함수를 설명합니다.

힌트

효율성, 안전성 및 이식성의 이유로 가능하면 워프 연산을 수행할 때 CUB Warp-Wide “Collective” Primitives 를 사용하는 것이 제안됩니다.

5.4.6.1. 워프 활성 마스크#

unsigned __activemask();

이 함수는 호출 워프에서 현재 활성 상태인 모든 스레드를 나타내는 32비트 정수 마스크를 반환합니다. 워프의 N번째 레인이 __activemask()가 호출될 때 활성 상태이면 N번째 비트가 설정됩니다. 비활성 스레드는 반환된 마스크에서 0 비트로 표현됩니다. 프로그램을 종료한 스레드는 항상 비활성으로 표시됩니다.

경고

__activemask()는 어떤 워프 레인이 주어진 분기를 실행하는지 결정하는 데 사용할 수 없습니다. 이 함수는 기회주의적 워프 수준 프로그래밍을 위한 것이며, 워프 내 활성 스레드의 순간적인 스냅샷만 제공합니다.

// Check whether at least one thread's predicate evaluates to true
if (pred) {
    // Invalid: the value of 'at_least_one' is non-deterministic
    // and could vary between executions.
    at_least_one = __activemask() > 0;
}

__activemask() 호출에서 수렴(convergent)한 스레드는, 그 이후의 명령이 워프 동기화 내장 함수(__sync)가 아닌 한, 후속 명령에서도 수렴 상태를 유지한다고 보장되지 않습니다.

예를 들어 컴파일러가 명령을 재정렬할 수 있으며, 활성 스레드 집합이 보존되지 않을 수 있습니다:

unsigned mask      = __activemask();              // Assume mask == 0xFFFFFFFF (all bits set, all threads active)
int      predicate = threadIdx.x % 2 == 0;        // 1 for even threads, 0 for odd threads
int      result    = __any_sync(mask, predicate); // Active threads might not be preserved

5.4.6.2. 워프 투표 함수#

int      __all_sync   (unsigned mask, int predicate);
int      __any_sync   (unsigned mask, int predicate);
unsigned __ballot_sync(unsigned mask, int predicate);

워프 투표 함수는 지정된 warp의 스레드가 축소(reduction)-및-브로드캐스트 연산을 수행할 수 있도록 합니다. 이러한 함수는 워프에서 종료되지 않은 각 스레드로부터 정수 predicate를 입력으로 받아 해당 값들을 0과 비교합니다. 그런 다음 비교 결과는 다음 방식 중 하나로 워프의 활성 스레드 전반에서 결합(축소)되며, 참여하는 각 스레드에 단일 반환 값을 브로드캐스트합니다:

__all_sync(unsigned mask, predicate):

평가합니다 predicate 워프에서 종료되지 않은 모든 스레드에 대해 mask 그리고 다음 경우 0이 아닌 값을 반환합니다 predicate 그들 모두에 대해 0이 아닌 값으로 평가되는 경우입니다.

__any_sync(unsigned mask, predicate):

평가합니다 predicate 워프에서 종료되지 않은 모든 스레드에 대해 mask 그리고 다음인 경우 0이 아닌 값을 반환합니다 predicate 그중 하나 이상에 대해 0이 아닌 값으로 평가되는 경우.

__ballot_sync(unsigned mask, predicate):

평가합니다 predicate 워프에서 종료되지 않은 모든 스레드에 대해 mask 그리고 워프의 N번째 스레드에 대해 predicate가 0이 아닌 값으로 평가되고 N번째 스레드가 활성 상태인 경우 N번째 비트가 설정된 정수를 반환합니다. 그렇지 않으면 N번째 비트는 0입니다.

이 함수들은 워프 __sync 내장 함수 제약 조건의 적용을 받습니다.

5.4.6.5. 워프 셔플 함수#

힌트

libcu++ cuda::device::warp_shuffle() 함수를 __shfl_sync() 및 __shfl_<op>_sync() 내장 함수(intrinsics)에 대한 일반화되고 더 안전한 대안으로 사용하는 것을 제안합니다.

T __shfl_sync     (unsigned mask, T value, int      srcLane,  int width=warpSize);
T __shfl_up_sync  (unsigned mask, T value, unsigned delta,    int width=warpSize);
T __shfl_down_sync(unsigned mask, T value, unsigned delta,    int width=warpSize);
T __shfl_xor_sync (unsigned mask, T value, int      laneMask, int width=warpSize);

워프 셔플 함수는 워프 내에서 종료되지 않은 스레드 간에 값을 교환합니다. 워프 공유 메모리를 사용하지 않고.

__shfl_sync(): 인덱싱된 레인에서 직접 복사합니다.

이 내장 함수(intrinsic function)는 다음의 값을 반환합니다 value 다음에 의해 ID가 지정된 스레드가 보유한 srcLane.

  • 만약 width 보다 작으면 warpSize, 그러면 워프의 각 하위 섹션은 시작 논리 레인 ID가 0인 별도의 엔티티로 동작합니다.

  • 만약 srcLane가 범위 [0, width - 1] 밖에 있으면, 결과는 동일한 하위 섹션 내에 있는 srcLane % width가 보유한 값에 해당합니다.

__shfl_up_sync(): 호출자보다 ID가 낮은 레인에서 복사합니다.

이 내장 함수(intrinsic function)는 호출자의 레인 ID에서 delta를 빼서 소스 레인 ID를 계산합니다. 그 결과 레인 ID가 보유한 value 값이 반환됩니다. 즉, value는 워프에서 delta개의 레인만큼 위로 시프트됩니다.

  • 만약 width가 warpSize보다 작으면, 워프의 각 하위 섹션은 시작 논리 레인 ID가 0인 별도의 엔티티로 동작합니다.

  • 소스 레인 인덱스는 width 값으로 래핑되지 않으므로, 하위 delta개 레인은 변경되지 않은 채로 유지됩니다.

__shfl_down_sync(): 호출자보다 ID가 높은 레인에서 복사합니다.

이 내장 함수(intrinsic function)는 호출자의 레인 ID에 delta를 더하여 소스 레인 ID를 계산합니다. 그 결과 레인 ID가 보유한 value 값이 반환됩니다. 이는 워프 내에서 value를 delta개 레인만큼 아래로 시프트하는 효과가 있습니다.

  • 만약 width가 warpSize보다 작으면, 워프의 각 하위 섹션은 시작 논리 레인 ID가 0인 별도의 엔티티로 동작합니다.

  • __shfl_up_sync()의 경우와 마찬가지로, 소스 레인의 ID 번호는 width 값으로 래핑되지 않으므로 상위 delta개 레인은 사실상 변경되지 않은 채로 유지됩니다.

__shfl_xor_sync(): 자신의 레인 ID의 비트 단위 XOR에 기반하여 레인에서 복사합니다.

이 내장 함수(intrinsic function)는 호출자의 레인 ID와 laneMask의 비트 단위 XOR을 수행하여 소스 레인 ID를 계산합니다. 그 결과 레인 ID가 보유한 value 값이 반환됩니다. 이 모드는 버터플라이 주소 지정 패턴을 구현하며, 이는 트리 축소(reduction) 및 브로드캐스트에 사용됩니다.

  • 만약 width가 warpSize보다 작으면, 연속된 스레드 width개로 이루어진 각 그룹은 이전 그룹의 요소에 접근할 수 있습니다. 그러나 이후 스레드 그룹의 요소에 접근을 시도하면, 자신의 value 값이 반환됩니다.

T는 다음 중 하나일 수 있습니다:

  • int, unsigned, long, unsigned long, long long, unsigned long long, float 또는 double.

  • __half 그리고 __half2 다음과 함께 cuda_fp16.h 헤더가 포함됩니다.

  • __nv_bfloat16 및 __nv_bfloat162는 cuda_bf16.h 헤더가 포함된 상태로 제공됩니다.

스레드는 내장 함수(intrinsics)에 적극적으로 참여하고 있는 다른 스레드로부터만 데이터를 읽을 수 있습니다. 대상 스레드가 비활성 상태이면, 가져온 값은 정의되지 않습니다.

width는 [1, warpSize] 범위의 2의 거듭제곱이어야 하며, 즉 1, 2, 4, 8, 16 또는 32여야 합니다. 다른 값은 정의되지 않은 결과를 생성합니다.

이 함수들은 Warp __sync Intrinsic Constraints의 적용을 받습니다.

유효한 워프 셔플 사용 예:

int laneId = threadIdx.x % warpSize;
int data   = ...

// all warp threads get 'data' from lane 0
int result1 = __shfl_sync(0xFFFFFFFF, data, 0);

if (laneId < 4) {
    // lanes 0, 1, 2, 3 get 'data' from lane 1
    int result2 = __shfl_sync(0xb1111, data, 1);
}

// lanes [0 - 15] get 'data' from lane 0
// lanes [16 - 31] get 'data' from lane 16
int result3 = __shfl_sync(0xFFFFFFFF, value, warpSize / 2);

// each lane gets 'data' from the lane two positions above
// lanes 30, 31 get their original value
int result4 = __shfl_down_sync(0xFFFFFFFF, data, 2);

유효하지 않은 워프 셔플 사용 예:

int laneId = threadIdx.x % warpSize;
int value  = ...
 // undefined behavior: lane 0 does not participate in the call
int result = (laneId > 0) ? __shfl_sync(0xFFFFFFFF, value, 0) : 0;

if (laneId <= 4) {
    // undefined behavior: destination lanes 5, 6 are not active for lanes 3, 4
    result = __shfl_down_sync(0b11111, value, 2);
}

// undefined behavior: width is not a power of 2
__shfl_sync(0xFFFFFFFF, value, 0, /*width=*/31);

예제 2: 8개 스레드의 하위 파티션 전반에서의 포함(inclusive) 덧셈 스캔

힌트

#include <cub/cub.cuh>

    using TempStorage = typename WarpScan::TempStorage;
    __shared__ TempStorage temp_storage;

    int laneId = threadIdx.x % 32;
    int value  = 31 - laneId; // starting value to accumulate
    int partial_sum;
    WarpScan(temp_storage).InclusiveSum(value, partial_sum);
    printf("Thread %d final value = %d\n", threadIdx.x, partial_sum);
}

    scan_sub_partition_with_8_threads_kernel<<<1, 32>>>();
    cudaDeviceSynchronize();
    return 0;
}

#include <stdio.h>

__global__ void scan_sub_partition_with_8_threads_kernel() {
    int laneId = threadIdx.x % 32;
    int value  = 31 - laneId; // starting value to accumulate
    // Loop to accumulate scan within my partition.
    // Scan requires log2(8) == 3 steps for 8 threads
    for (int delta = 1; delta <= 4; delta *= 2) {
        int tmp         = __shfl_up_sync(0xFFFFFFFF, value, delta, /*width=*/8); // read from laneId - delta
        int source_lane = laneId % 8 - delta;
        if (source_lane >= 0) // lanes with 'source_lane < 0' have their value unchanged
            value += tmp;
    }
    printf("Thread %d final value = %d\n", threadIdx.x, value);
}

int main() {
    scan_sub_partition_with_8_threads_kernel<<<1, 32>>>();
    cudaDeviceSynchronize();
    return 0;
}

예제 3: 워프 전반의 축소(reduction)

힌트

효율적이고 일반화된 워프 축소 함수에는 cub::WarpReduce 함수를 사용하는 것이 제안됩니다.

#include <cstdio>
#include <cub/cub.cuh>
#include <cuda/warp>

__global__ void warp_reduce_kernel() {
    using WarpReduce  = cub::WarpReduce<int>;
    using TempStorage = typename WarpReduce::TempStorage;
    __shared__ TempStorage temp_storage;

    int laneId     = threadIdx.x % 32;
    int value      = 31 - laneId; // starting value to accumulate
    auto aggregate = WarpReduce(temp_storage).Sum(value);

}

}

#include <stdio.h>

__global__ void warp_reduce_kernel() {
    int laneId = threadIdx.x % 32;
    int value  = 31 - laneId; // starting value to accumulate
    // Use XOR mode to perform butterfly reduction
    // A full-warp reduction requires log2(32) == 5 steps
    for (int i = 1; i <= 16; i *= 2)

    printf("Thread %d final value = %d\n", threadIdx.x, value);
}

int main() {
    warp_reduce_kernel<<<1, 32>>>();
    cudaDeviceSynchronize();
    return 0;
}

5.4.6.6. Warp __sync 내장 함수(intrinsics) 제약 조건#

다음과 같은 모든 워프 __sync 내장 함수(intrinsics)는:

  • __shfl_sync, __shfl_up_sync, __shfl_down_sync, __shfl_xor_sync

  • __match_any_sync, __match_all_sync

  • __reduce_add_sync, __reduce_min_sync, __reduce_max_sync, __reduce_and_sync, __reduce_or_sync, __reduce_xor_sync

  • __syncwarp

호출에 어떤 워프 스레드가 참여하는지 나타내기 위해 mask 매개변수를 사용합니다. 이 매개변수는 하드웨어가 내장 함수를 실행하기 전에 적절한 컨버전스(convergence)를 보장합니다.

mask의 각 비트는 스레드의 레인 ID(threadIdx.x % warpSize)에 대응합니다. 내장 함수는 mask에 지정된 종료되지 않은 모든 워프 스레드가 해당 호출 지점에 도달할 때까지 대기합니다.
올바른 실행을 위해서는 다음 제약 조건을 충족해야 합니다:

  • 각 호출 스레드는 mask에서 해당 비트가 설정되어 있어야 합니다.

  • 각 비호출 스레드는 mask에서 해당 비트가 0으로 설정되어 있어야 합니다. 종료된 스레드는 무시됩니다.

  • mask에 지정된 종료되지 않은 모든 스레드는 동일한 mask 값으로 내장 함수를 실행해야 합니다.

  • 워프 스레드는 마스크가 서로 겹치지 않는(disjoint) 경우 서로 다른 mask 값으로 동시에 내장 함수를 호출할 수 있습니다. 이러한 조건은 제어 흐름이 발산(divergent)하더라도 유효합니다.

워프 __sync 함수의 동작은 다음과 같은 경우 커널 행(hang) 등으로 인해 잘못되거나, 정의되지 않습니다:

  • 호출 스레드가 mask에 지정되지 않은 경우.

  • mask에 지정된 종료되지 않은 스레드가 결국 종료하거나, 동일한 프로그램 지점에서 동일한 mask 값으로 내장 함수를 호출하는 것 중 어느 하나도 수행하지 못하는 경우.

  • 조건부 코드에서는, mask에 지정된 종료되지 않은 모든 스레드 전반에서 모든 조건이 동일하게 평가되어야 합니다.

Note

이 내장 함수(intrinsics)는 모든 워프 스레드가 호출에 참여할 때, 즉 mask가 0xFFFFFFFF로 설정될 때 가장 높은 효율을 달성합니다.

유효한 워프 내장 함수(intrinsics) 사용 예:

__global__ void valid_examples() {
    if (threadIdx.x < 4) {        // threads 0, 1, 2, 3 are active
        __all_sync(0b1111, pred); // CORRECT, threads 0, 1, 2, 3 participate in the call
    }

    if (threadIdx.x == 0)
        return; // exit
    // CORRECT, all non-exited threads participate in the call
    __all_sync(0xFFFFFFFF, pred);
}

서로 겹치지 않는(disjoint) mask 예:

__global__ void example_syncwarp_with_mask(int* input_data, int* output_data) {
    if (threadIdx.x < warpSize) {
        __shared__ int shared_data[warpSize];
        shared_data[threadIdx.x] = input_data[threadIdx.x];

        unsigned mask = threadIdx.x < 16 ? 0xFFFF : 0xFFFF0000; // CORRECT
        __syncwarp(mask);
        if (threadIdx.x == 0 || threadIdx.x == 16)
            output_data[threadIdx.x] = shared_data[threadIdx.x + 1];
    }
}

__global__ void example_syncwarp_with_mask_branches(int* input_data, int* output_data) {
    if (threadIdx.x < warpSize) {
        __shared__ int shared_data[warpSize];
        shared_data[threadIdx.x] = input_data[threadIdx.x];

        if (threadIdx.x < 16) {
            unsigned mask = 0xFFFF; // CORRECT
            __syncwarp(mask);
            output_data[threadIdx.x] = shared_data[15 - threadIdx.x];
        }
        else {
            unsigned mask = 0xFFFF0000; // CORRECT
            __syncwarp(mask);
            output_data[threadIdx.x] = shared_data[31 - threadIdx.x];
        }
    }
}

유효하지 않은 워프 내장 함수(intrinsics) 사용 예:

if (threadIdx.x < 4) {           // threads 0, 1, 2, 3 are active
    __all_sync(0b0000011, pred); // WRONG, threads 2, 3 are active but not set in mask
    __all_sync(0b1111111, pred); // WRONG, threads 4, 5, 6 are not active but set in mask
}

// WRONG, participating threads have a different and overlapping mask
__all_sync(threadIdx.x == 0 ? 1 : 0xFFFFFFFF, pred);

5.4.7. CUDA 특정 매크로#

5.4.7.1. __CUDA_ARCH__#

매크로 __CUDA_ARCH__는 코드가 컴파일되는 대상인 NVIDIA GPU의 가상 아키텍처를 나타냅니다. 그 값은 디바이스의 실제 컴퓨트 기능과 다를 수 있습니다. 이 매크로를 사용하면 특정 GPU 아키텍처에 특화된 코드 경로를 작성할 수 있으며, 이는 최적의 성능을 위해 필요하거나 아키텍처별 기능 및 명령을 사용하기 위해 필요할 수 있습니다. 또한 이 매크로는 호스트 코드와 디바이스 코드를 구분하는 데 사용할 수 있습니다.

__CUDA_ARCH__는 디바이스 코드에서만, 즉 __device__, __host__ __device__ 및 __global__ 함수에서만 정의됩니다. 이 매크로의 값은 nvcc 옵션 compute_<version>와 연관되며, 그 관계는 __CUDA_ARCH__ = <version> * 10입니다.

예:

nvcc --generate-code arch=compute_80,code=sm_90 prog.cu

__CUDA_ARCH__를 800로 정의합니다.

__CUDA_ARCH__ 제약 조건

1. 다음 엔티티의 타입 시그니처는 __CUDA_ARCH__의 정의 여부나 그 값에 의존해서는 안 됩니다.

  • __global__ 함수 및 함수 템플릿.

  • __device__ 및 __constant__ 변수.

  • 텍스처 및 서피스.

예:

#if !defined(__CUDA_ARCH__)
    typedef int my_type;
#else
    typedef double my_type;
#endif

__device__ my_type my_var;           // ERROR: my_var's type depends on __CUDA_ARCH__

__global__ void kernel(my_type in) { // ERROR: kernel's type depends on __CUDA_ARCH__
    ...
}

2. __global__ 함수 템플릿이 인스턴스화되어 호스트에서 실행(launch)되는 경우, __CUDA_ARCH__의 정의 여부나 그 값과 무관하게 동일한 템플릿 인자를 사용해 인스턴스화되어야 합니다.

예:

__device__ int result;

template <typename T>
__global__ void kernel(T in) {
    result = in;
}

__host__ __device__ void host_device_function(void) {
#if !defined(__CUDA_ARCH__)
    kernel<<<1, 1>>>(1); // ERROR: "kernel<int>" instantiation only
                            //        when __CUDA_ARCH__ is undefined!
#endif
}

int main(void) {
    host_device_function();
    cudaDeviceSynchronize();
    return 0;
}

3. 분리 컴파일 모드에서는 외부 링키지(external linkage)를 가진 함수 또는 변수 정의의 존재 여부가 __CUDA_ARCH__의 정의나 그 값에 의존해서는 안 됩니다.

예:

#if !defined(__CUDA_ARCH__)
    void host_function(void) {} // ERROR: The definition of host_function()
                                //        is only present when __CUDA_ARCH__
                                //        is undefined
#endif

4. 분리 컴파일에서는 전처리기 매크로 __CUDA_ARCH__를 헤더에서 사용하여 객체들이 서로 다른 동작을 갖도록 해서는 안 됩니다. 대안으로, 모든 객체는 동일한 가상 아키텍처를 대상으로 컴파일되어야 합니다. 약한(weak) 함수 또는 템플릿 함수가 헤더에 정의되어 있고 그 동작이 __CUDA_ARCH__에 의존한다면, 서로 다른 컴퓨트 아키텍처를 대상으로 컴파일된 서로 다른 객체에서 해당 함수의 인스턴스들이 충돌할 수 있습니다.

예를 들어, 헤더 파일 a.h에 다음이 포함되어 있다고 가정합니다:

template<typename T>
__device__ T* get_ptr() {
#if __CUDA_ARCH__ == 900
    return nullptr; /* no address */
#else
    __shared__ T arr[256];
    return arr;
#endif
}

그러면 a.cu 및 b.cu가 모두 a.h를 포함하고 동일한 타입에 대해 get_ptr()를 인스턴스화하며, b.cu가 비-NULL 주소를 기대하고, 다음과 같이 컴파일한다면:

nvcc -arch=compute_70 -dc a.cu
nvcc -arch=compute_80 -dc b.cu
nvcc -arch=sm_80 a.o b.o

Only one version of the ``get_ptr()`` function is used at link time, so the behavior depends on which version is chosen. To avoid this issue, either ``a.cu`` and ``b.cu`` must be compiled for the same compute architecture, or ``__CUDA_ARCH__`` should not be used in the shared header function.

컴파일러는 위에서 설명한 __CUDA_ARCH__의 지원되지 않는 사용에 대해 진단 메시지가 생성될 것을 보장하지 않습니다.

5.4.7.2. __CUDA_ARCH_SPECIFIC__ 및 __CUDA_ARCH_FAMILY_SPECIFIC__#

매크로 __CUDA_ARCH_SPECIFIC__ 및 __CUDA_ARCH_FAMILY_SPECIFIC__는 각각 architecture- 및 family- 특정 기능을 가진 GPU 장치를 식별하도록 정의되어 있습니다. 자세한 내용은 Feature Set Compiler Targets 섹션을 참조하십시오.

__CUDA_ARCH__와 유사하게, __CUDA_ARCH_SPECIFIC__ 및 __CUDA_ARCH_FAMILY_SPECIFIC__는 디바이스 코드에서만, 즉 __device__, __host__ __device__, 그리고 __global__ 함수에서만 정의됩니다. 이 매크로들은 nvcc 옵션인 compute_<version>a 및 compute_<version>f와 연관되어 있습니다.

nvcc --generate-code arch=compute_100a,code=sm_100a prog.cu

  • __CUDA_ARCH__ == 1000.

  • __CUDA_ARCH_SPECIFIC__ == 1000.

  • __CUDA_ARCH_FAMILY_SPECIFIC__ == 1000.

nvcc --generate-code arch=compute_100f,code=sm_103f prog.cu

  • __CUDA_ARCH__ == 1000.

  • __CUDA_ARCH_FAMILY_SPECIFIC__ == 1000.

  • __CUDA_ARCH_SPECIFIC__는 정의되지 않습니다.

nvcc -arch=sm_100 prog.cu

  • __CUDA_ARCH__ == 1000.

  • __CUDA_ARCH_FAMILY_SPECIFIC__는 정의되지 않습니다.

  • __CUDA_ARCH_SPECIFIC__는 정의되지 않습니다.

nvcc -arch=sm_100a prog.cu
# equivalent to:
nvcc --generate-code arch=sm_100a,compute_100,compute_100a prog.cu

  • __CUDA_ARCH__ == 1000.

  • __CUDA_ARCH_FAMILY_SPECIFIC__는 정의되지 않습니다.

  • __CUDA_ARCH_SPECIFIC__ == 1000 및 __CUDA_ARCH_SPECIFIC__ not defined가 모두 생성됩니다.

5.4.7.4. __nv_pure__ 속성#

C/C++에서 순수 함수(pure function)는 매개변수에 대한 부작용이 없으며 전역 변수에 접근할 수 있지만, 이를 수정하지는 않습니다.

CUDA는 호스트 및 디바이스 함수 모두에 대해 지원되는 __nv_pure__ 속성을 제공합니다. 컴파일러는 __nv_pure__를 pure GNU 속성 또는 Microsoft Visual Studio noalias 속성으로 변환합니다.

__device__ __nv_pure__
int add(int a, int b) {
    return a + b;
}

힌트

5.4.8.2. 주소 공간 변환 함수#

CUDA 포인터(T*)는 객체가 저장된 위치와 무관하게 객체에 접근할 수 있습니다. 예를 들어, int*는 전역 또는 공유 메모리에 상주하는지 여부와 관계없이 int 객체에 접근할 수 있습니다.

주소 공간 변환 함수는 제네릭 주소와 특정 주소 공간의 주소 간 변환에 사용됩니다. 이 함수들은 컴파일러가 포인터의 주소 공간을 결정할 수 없는 경우에 유용합니다. 예를 들어, 번역 단위(translation unit)를 넘나들거나 PTX 명령과 상호작용할 때입니다.

__device__ size_t __cvta_generic_to_global  (const void* ptr); // PTX: cvta.to.global
__device__ size_t __cvta_generic_to_shared  (const void* ptr); // PTX: cvta.to.shared
__device__ size_t __cvta_generic_to_constant(const void* ptr); // PTX: cvta.to.const
__device__ size_t __cvta_generic_to_local   (const void* ptr); // PTX: cvta.to.local

__device__ void* __cvta_global_to_generic  (size_t raw_ptr); // PTX: cvta.global
__device__ void* __cvta_shared_to_generic  (size_t raw_ptr); // PTX: cvta.shared
__device__ void* __cvta_constant_to_generic(size_t raw_ptr); // PTX: cvta.const
__device__ void* __cvta_local_to_generic   (size_t raw_ptr); // PTX: cvta.local

PTX 명령과 상호 운용하는 예로, ld.shared.s32 r0, [ptr]; PTX 명령은 다음을 기대합니다 ptr 공유 메모리 주소 공간을 참조하기 위해서입니다. A CUDA 프로그램에서 int* 내의 객체에 대한 포인터 __shared__ 메모리에서 이 포인터를 PTX 명령에 전달하기 전에, 호출을 통해 이 포인터를 공유 주소 공간으로 변환해야 합니다 __cvta_generic_to_shared 다음과 같습니다:

__shared__ int smem_var;
smem_var        = 42;
size_t smem_ptr = __cvta_generic_to_shared(&smem_var);
int    output;
asm volatile("ld.shared.s32 %0, [%1];" : "=r"(output) : "l"(smem_ptr) : "memory");
assert(output == 42);

이러한 주소 표현을 활용하는 일반적인 최적화는 공유, 로컬 및 상수 공간의 주소 범위가 32비트보다 작다는 사실을 이용하여 데이터 구조의 크기를 줄이는 것입니다. 이를 통해 64비트 포인터 대신 32비트 주소를 저장할 수 있으며 레지스터를 절약할 수 있습니다. 또한 32비트 산술 연산은 64비트 산술 연산보다 더 빠릅니다. 이러한 주소의 32비트 정수 표현을 얻으려면, 부호 없는 64비트 정수에서 부호 없는 32비트 정수로 캐스팅하여 64비트 값을 32비트로 절단(truncate)하십시오:

__shared__ int smem_var;
uint32_t       smem_ptr_32bit = static_cast<uint32_t>(__cvta_generic_to_shared(&smem_var));

이러한 32비트 표현으로부터 일반 주소를 복원하려면, 주소를 부호 없는 64비트 정수로 다시 0-확장(zero-extend)한 다음 해당 주소 공간 변환 함수를 호출하십시오:

size_t smem_ptr_64bit = static_cast<size_t>(smem_ptr_32bit); // zero-extend to 64 bits
void*  generic_ptr    = __cvta_shared_to_generic(smem_ptr_64bit);
assert(generic_ptr == &smem_var);

5.4.8.4. __trap()#

힌트

다음을 사용하는 것이 권장됩니다 cuda::std::terminate() 에서 제공하는 함수 libcu++ (C++ 레퍼런스) 이식 가능한 대안으로 __trap().

트랩(trap) 연산은 다음을 호출하여 시작할 수 있습니다. __trap() 어떤 디바이스 스레드에서든 해당 함수를 호출하여 시작할 수 있습니다.

void __trap();

커널 실행이 중단되며, 호스트 프로그램에서 인터럽트가 발생합니다. __trap()를 호출하면 CUDA 컨텍스트가 손상되어, 이후의 CUDA 호출 및 커널 호출이 실패하게 됩니다.

5.4.8.5. __nanosleep()#

__device__ void __nanosleep(unsigned nanoseconds);

함수 __nanosleep(ns)는 스레드를 약 ns 나노초 동안 슬립(sleep) 지속 시간만큼 일시 중지합니다. 최대 슬립 지속 시간은 약 1밀리초입니다.

예:

다음 코드는 지수 백오프(exponential back-off)를 사용하는 뮤텍스를 구현합니다.

__device__ void mutex_lock(unsigned* mutex) {
    unsigned ns = 8;
    while (atomicCAS(mutex, 0, 1) == 1) {
        __nanosleep(ns);
        if (ns < 256) {
            ns *= 2;
        }
    }
}

__device__ void mutex_unlock(unsigned *mutex) {
    atomicExch(mutex, 0);
}

5.4.9. 컴파일러 최적화 힌트#

컴파일러 최적화 힌트는 생성된 코드를 컴파일러가 최적화하는 데 도움이 되도록 추가 정보로 코드를 데코레이션합니다.

  • 내장 함수(intrinsics)는 디바이스 코드에서 항상 사용할 수 있습니다.

  • 호스트 코드 지원은 호스트 컴파일러에 따라 달라집니다.

5.4.9.1. #pragma unroll#

컴파일러는 기본적으로 반복 횟수(trip count)가 알려진 작은 루프를 언롤(unroll)합니다. 그러나 #pragma unroll 지시문을 사용하여 특정 루프의 언롤링을 제어할 수 있습니다. 이 지시문은 루프 바로 앞에 즉시 배치되어야 하며, 해당 루프에만 적용됩니다.

정수 상수 표현식이 선택적으로 뒤따를 수 있습니다. 다음은 정수 상수 표현식에 대한 경우입니다:

  • 표현식이 없으면, 반복 횟수(trip count)가 상수인 경우 루프는 완전히 언롤(unroll)됩니다.

  • 표현식의 평가 결과가 0 또는 1이면, 루프는 언롤되지 않습니다.

  • 표현식이 0 이하의 정수이거나 INT_MAX보다 크면, pragma는 무시되며 경고가 발생합니다.

예:

struct MyStruct {
    static constexpr int value = 4;
};

inline constexpr int Count = 4;

__device__ void foo(int* p1, int* p2) {
    // no argument specified, the loop will be completely unrolled
    #pragma unroll
    for (int i = 0; i < 12; ++i)
        p1[i] += p2[i] * 2;

    // unroll value = 5
    #pragma unroll (Count + 1)
    for (int i = 0; i < 12; ++i)
        p1[i] += p2[i] * 4;

    // unroll value = 1, loop unrolling disabled
    #pragma unroll 1
    for (int i = 0; i < 12; ++i)
        p1[i] += p2[i] * 8;

    // unroll value = 4
    #pragma unroll (MyStruct::value)
    for (int i = 0; i < 12; ++i)
        p1[i] += p2[i] * 16;

    // negative value, pragma unroll ignored
    #pragma unroll -1
    for (int i = 0; i < 12; ++i)
        p1[i] += p2[i] * 2;
}

5.4.9.2. __builtin_assume_aligned()#

힌트

다음을 사용하는 것이 권장됩니다. cuda::std::assume_aligned() 다음에서 제공하는 함수 libcu++ (C++ 레퍼런스) 내장 함수에 대한 이식 가능하고 더 안전한 대안으로 사용할 수 있습니다.

void* __builtin_assume_aligned(const void* ptr, size_t align)
void* __builtin_assume_aligned(const void* ptr, size_t align, <integral type> offset)

이 빌트인 함수들은 컴파일러가 반환된 포인터가 최소한 다음과 같이 정렬되어 있다고 가정할 수 있게 합니다 align 바이트입니다.

  • 세 개 매개변수 버전은 컴파일러가 다음을 가정할 수 있게 합니다 (char*) ptr - offset 최소한 다음에 정렬되어 있다고 align 바이트입니다.

align는 2의 거듭제곱이며 정수 리터럴이어야 합니다.

예:

void* res1 = __builtin_assume_aligned(ptr, 32);    // compiler can assume 'res1' is at least 32-byte aligned
void* res2 = __builtin_assume_aligned(ptr, 32, 8); // compiler can assume 'res2 = (char*) ptr - 8' is at least 32-byte aligned

5.4.9.3. __builtin_assume() 및 __assume()#

void __builtin_assume(bool predicate)
void __assume        (bool predicate) // only with Microsoft Compiler

이 내장 함수(built-in function)는 컴파일러가 불리언 인수가 true라고 가정할 수 있게 합니다. 런타임에 인수가 false이면, 동작은 정의되지 않습니다. 또한 인수에 부작용이 있는 경우, 동작은 미지정(unspecified)임에 유의하십시오.

예:

__device__ bool is_greater_than_zero(int value) {
    return value > 0;
}

__device__ bool f(int value) {
    __builtin_assume(value > 0);
    return is_greater_than_zero(value); // returns true, without evaluating the condition
}

5.4.9.4. __builtin_expect()#

long __builtin_expect(long input, long expected)

이 내장 함수는 컴파일러에 input가 expected와 같을 것으로 예상된다는 것을 알려 주며, input의 값을 반환합니다. 이는 일반적으로 컴파일러에 분기 예측 정보를 제공하는 데 사용됩니다. 이는 C++20의 [[likely]] 및 [[unlikely]] attributes처럼 동작합니다.

예:

// indicate to the compiler that likely "var == 0"
if (__builtin_expect(var, 0))
    doit();

5.4.9.5. __builtin_unreachable()#

void __builtin_unreachable(void)

이 내장 함수는 컴파일러에 제어 흐름이 함수가 호출되는 지점에 결코 도달하지 않을 것임을 알려 줍니다. 런타임에 제어 흐름이 이 지점에 도달하면, 프로그램은 정의되지 않은 동작을 합니다.

이 함수는 도달할 수 없는 분기에 대한 코드 생성을 피하고, 도달할 수 없는 코드에 대한 컴파일러 경고를 비활성화하는 데 유용합니다.

예:

// indicates to the compiler that the default case label is never reached.
switch (in) {
    case 1:  return 4;
    case 2:  return 10;
    default: __builtin_unreachable();
}

5.4.9.6. 사용자 정의 ABI 프라그마#

#pragma nv_abi 지시문은 함수가 사용하는 레지스터 수를 보존함으로써, 분리 컴파일 모드로 컴파일된 애플리케이션이 전체 프로그램 컴파일과 유사한 성능을 달성할 수 있게 합니다.

이 pragma를 사용하는 구문은 다음과 같으며, 여기서 EXPR는 임의의 정수 상수 표현식을 의미합니다:

#pragma nv_abi preserve_n_data(EXPR) preserve_n_control(EXPR)

  • #pragma nv_abi 뒤에 오는 인수는 선택 사항이며 어떤 순서로도 제공될 수 있습니다. 단, 최소 하나의 인수는 필요합니다.

  • preserve_n 인수는 함수 호출 동안 보존되는 레지스터 수를 제한합니다:

    • preserve_n_data(EXPR)는 데이터 레지스터 수를 제한합니다.

    • preserve_n_control(EXPR)는 제어 레지스터 수를 제한합니다.

#pragma nv_abi 지시문은 디바이스 함수 선언 또는 정의 바로 앞에 둘 수 있습니다.

#pragma nv_abi preserve_n_data(16)
__device__ void dev_func();

#pragma nv_abi preserve_n_data(16) preserve_n_control(8)
__device__ int dev_func() {
    return 0;
}

또는, 디바이스 함수 내부의 C++ 표현식 문에서 간접 함수 호출 바로 앞에 직접 둘 수 있습니다. 자유 함수에 대한 간접 함수 호출은 지원되지만, 함수 레퍼런스 또는 클래스 멤버 함수에 대한 간접 호출은 지원되지 않습니다.

__device__ int dev_func1();

struct MyStruct {
    __device__ int member_func2();
};

__device__ void test() {
    auto* dev_func_ptr = &dev_func1; // type: int (*)(void)
    #pragma nv_abi preserve_n_control(8)
    int v1 = dev_func_ptr();         // CORRECT, indirect call

    #pragma nv_abi preserve_n_control(8)
    int v2 = dev_func1();            // WRONG, direct call; the pragma has no effect
                                     // dev_func1 has type: int(void)

    auto& dev_func_ref = &dev_func1; // type: int (&)(void)
    #pragma nv_abi preserve_n_control(8)
    int v3 = dev_func_ref();         // WRONG, call to a reference
                                     // the pragma has no effect

    auto member_function_ptr = &MyStruct::member_func2; // type: int (MyStruct::*)(void)
    #pragma nv_abi preserve_n_control(8)
    int v4 = member_function_ptr();  // WRONG, indirect call to member function
                                     // the pragma has no effect
}

디바이스 함수의 선언 또는 정의에 적용되면, 이 pragma는 해당 함수에 대한 모든 호출의 사용자 정의 ABI 속성을 수정합니다. 간접 함수 호출 지점에 배치되면, 해당 특정 호출에 대해서만 ABI 속성에 영향을 줍니다. 또한 pragma는 호출 지점에 배치될 때에만 간접 함수 호출에 영향을 주며, 직접 함수 호출에는 아무런 영향을 주지 않습니다.

#pragma nv_abi preserve_n_control(8)
__device__ int dev_func3();

__device__ int dev_func4();

__device__ void test() {
    int v1 = dev_func3();            // CORRECT, the pragma affects the direct call

    auto* dev_func_ptr = &dev_func4; // type: int (*)(void)
    #pragma nv_abi preserve_n_control(8)
    int v2 = dev_func_ptr();         // CORRECT, the pragma affects the indirect call

    int v3 = dev_func_ptr();         // WRONG, the pragma has no effect
}

함수 선언과 이에 대응하는 정의에 대한 pragma 인수가 일치하지 않으면 프로그램은 형식상 올바르지 않음(ill-formed)임에 유의하십시오.

5.4.10. 디버깅 및 진단#

5.4.10.1. 단언(Assertion)#

void assert(int expression);

assert() 매크로는 expression가 0과 같으면 커널 실행을 중지합니다. 프로그램이 디버거 내에서 실행 중이면 중단점이 트리거되어, 디버거를 사용해 디바이스의 현재 상태를 검사할 수 있습니다. 그렇지 않으면, expression가 0과 같은 각 스레드는 cudaDeviceSynchronize(), cudaStreamSynchronize(), 또는 cudaEventSynchronize()를 통해 호스트와 동기화한 후 stderr에 메시지를 출력합니다. 이 메시지의 형식은 다음과 같습니다:

<filename>:<line number>:<function>:
block: [blockIdx.x,blockIdx.y,blockIdx.z],
thread: [threadIdx.x,threadIdx.y,threadIdx.z]
Assertion `<expression>` failed.

커널의 실행은 중단되며, 호스트 프로그램에서 인터럽트가 발생합니다. assert() 매크로는 손상된 CUDA 컨텍스트를 초래하여, 이후의 모든 CUDA 호출 또는 커널 호출이 cudaErrorAssert로 실패하게 합니다.

expression가 0이 아니면 커널 실행은 영향을 받지 않습니다.

예를 들어, 소스 파일 test.cu의 다음 프로그램은

#include <assert.h>

 __global__ void testAssert(void) {
     int is_one        = 1;
     int should_be_one = 0;

     // This will have no effect
     assert(is_one);

     // This will halt kernel execution
     assert(should_be_one);
 }

 int main(void) {
     testAssert<<<1,1>>>();
     cudaDeviceSynchronize();
     return 0;
 }

다음을 출력합니다:

test.cu:11: void testAssert(): block: [0,0,0], thread: [0,0,0] Assertion `should_be_one` failed.

단언(Assertion)은 디버깅 목적을 위한 것입니다. 이는 성능에 영향을 줄 수 있으므로, 프로덕션 코드에서는 비활성화할 것을 권장합니다. 이는 컴파일 시간에 다음을 정의하여 비활성화할 수 있습니다 NDEBUG 포함하기 전에 전처리기 매크로를 assert.h 또는 <cassert>, 또는 컴파일러 플래그를 사용하여 -DNDEBUG. 식에는 부작용이 없어야 한다는 점에 유의하십시오. 그렇지 않으면 단언을 비활성화할 때 코드의 기능에 영향을 미칩니다.

5.4.10.3. 진단 프라그마#

다음 pragma는 특정 진단 메시지가 발생할 때 트리거되는 오류의 심각도를 관리하는 데 사용할 수 있습니다.

#pragma nv_diag_suppress
#pragma nv_diag_warning
#pragma nv_diag_error
#pragma nv_diag_default
#pragma nv_diag_once

이 pragma의 사용법은 다음과 같습니다:

#pragma nv_diag_xxx <error_number1>, <error_number2> ...

영향을 받는 진단은 경고 메시지에 표시된 오류 번호를 사용하여 지정합니다. 어떤 진단이든 오류로 변경할 수 있지만, 오류로 변경된 후에는 경고만 그 심각도를 억제하거나 복원할 수 있습니다. The nv_diag_default pragma는 진단의 심각도를 다른 어떤 pragma도 발행되기 전에 적용되던 심각도, 즉 명령줄 옵션에 의해 수정된 메시지의 정상 심각도로 되돌립니다. 다음 예는 억제합니다 declared but never referenced 경고의 foo():

#pragma nv_diag_suppress 177 // "declared but never referenced"
void foo() {
    int i = 0;
}

#pragma nv_diag_default 177
void bar() {
    int i = 0;
}

다음 pragma는 현재 진단 pragma 상태를 저장하고 복원하는 데 사용할 수 있습니다:

#pragma nv_diagnostic push
#pragma nv_diagnostic pop

예:

#pragma nv_diagnostic push
#pragma nv_diag_suppress 177 // "declared but never referenced"
void foo() {
    int i = 0;
}

#pragma nv_diagnostic pop
void bar() {
    int i = 0; // raise a warning
}

이 지시문들은 nvcc CUDA 프런트엔드 컴파일러에만 영향을 미친다는 점에 유의하십시오. 호스트 컴파일러에는 아무런 영향이 없습니다.

nvcc는 진단 pragma가 지원될 때 매크로 __NVCC_DIAG_PRAGMA_SUPPORT__를 정의합니다.

5.4.11. 워프 행렬 함수#

C++ 워프 행렬 연산은 텐서 코어(Tensor Cores)를 활용하여 D=A*B+C 형태의 행렬 문제를 가속합니다. 이러한 연산은 컴퓨트 기능(compute capability) 7.0 이상인 디바이스에서 혼합 정밀도 부동소수점 데이터에 대해 지원됩니다. 이를 위해 warp 내의 모든 스레드의 협력이 필요합니다. 또한 이러한 연산은 조건이 전체 warp에 걸쳐 동일하게 평가되는 경우에만 조건부 코드에서 허용되며, 그렇지 않으면 코드 실행이 멈출 가능성이 큽니다.

5.4.11.1. 설명#

이하의 모든 함수와 타입은 네임스페이스 nvcuda::wmma에 정의되어 있습니다. 서브바이트(sub-byte) 연산은 프리뷰로 간주되며, 즉 이를 위한 데이터 구조와 API는 변경될 수 있고 향후 릴리스와 호환되지 않을 수 있습니다. 이러한 추가 기능은 nvcuda::wmma::experimental 네임스페이스에 정의되어 있습니다.

template<typename Use, int m, int n, int k, typename T, typename Layout=void> class fragment;

void load_matrix_sync(fragment<...> &a, const T* mptr, unsigned ldm);
void load_matrix_sync(fragment<...> &a, const T* mptr, unsigned ldm, layout_t layout);
void store_matrix_sync(T* mptr, const fragment<...> &a, unsigned ldm, layout_t layout);
void fill_fragment(fragment<...> &a, const T& v);
void mma_sync(fragment<...> &d, const fragment<...> &a, const fragment<...> &b, const fragment<...> &c, bool satf=false);
fragment

워프의 모든 스레드에 걸쳐 분산된 행렬의 한 섹션을 포함하는 오버로드된 클래스입니다. 행렬 요소를 fragment 내부 스토리지로 매핑하는 방식은 미지정(unspecified)이며, 향후 아키텍처에서 변경될 수 있습니다.

템플릿 인수의 특정 조합만 허용됩니다. 첫 번째 템플릿 매개변수는 프래그먼트가 행렬 연산에 어떻게 참여할지를 지정합니다. Use에 대해 허용되는 값은 다음과 같습니다:

  • 프래그먼트가 첫 번째 곱셈 피연산자로 사용될 때는 matrix_a, A,

  • matrix_b 프래그먼트가 두 번째 곱셈 피연산자로 사용될 때는 B, 또는

  • accumulator 프래그먼트가 소스 또는 목적지 누산기(accumulators)로 사용될 때 (C 또는 D, 각각).

    m, n 및 k 크기는 곱셈-누산(multiply-accumulate) 연산에 참여하는 워프 전체 행렬 타일의 형태를 설명합니다. 각 타일의 차원은 그 역할에 따라 달라집니다. matrix_a의 경우 타일은 m x k 차원을 가지며; matrix_b의 경우 차원은 k x n이고, accumulator 타일은 m x n입니다.

    데이터 타입, T, 일 수 있습니다 double, float, __half, __nv_bfloat16, char, 또는 unsigned char 곱해지는 수(multiplicands) 및 double, float, int, 또는 __half 누산기(accumulator)의 경우. 문서화된 바와 같이 요소 타입 및 행렬 크기, 누산기(accumulator)와 피승수(multiplicand) 타입의 제한된 조합만 지원됩니다. Layout 매개변수는 다음에 대해 지정되어야 합니다 matrix_a 그리고 matrix_b 프래그먼트. row_major 또는 col_major 각각 행렬의 행 또는 열 내 요소가 메모리에서 연속적임을 나타냅니다. The Layout 에 대한 매개변수 accumulator 행렬은 기본값을 유지해야 합니다 void. 행 또는 열 레이아웃은 아래에 설명된 대로 누산기가 로드되거나 저장될 때에만 지정됩니다.

load_matrix_sync

모든 워프 레인이 load_matrix_sync에 도달할 때까지 대기한 다음, 메모리에서 행렬 프래그먼트 a를 로드합니다. mptr는 메모리 내 행렬의 첫 번째 요소를 가리키는 256비트 정렬 포인터여야 합니다. ldm는 연속된 행(행 우선 레이아웃의 경우) 또는 열(열 우선 레이아웃의 경우) 사이의 요소 단위 스트라이드를 설명하며, __half 요소 타입의 경우 8의 배수이거나 float 요소 타입의 경우 4의 배수여야 합니다. (즉, 두 경우 모두 16바이트의 배수). 프래그먼트가 accumulator인 경우, layout 인수는 mem_row_major 또는 mem_col_major 중 하나로 지정되어야 합니다. matrix_a 및 matrix_b 프래그먼트의 경우, 레이아웃은 프래그먼트의 layout 매개변수로부터 유추됩니다. mptr, ldm, layout의 값과 a에 대한 모든 템플릿 매개변수는 워프 내 모든 스레드에서 동일해야 합니다. 이 함수는 워프 내 모든 스레드가 호출해야 하며, 그렇지 않으면 결과는 미지정(unspecified)입니다.

store_matrix_sync

모든 워프 레인이 store_matrix_sync에 도달할 때까지 대기한 다음, 행렬 프래그먼트 a를 메모리에 저장합니다. mptr는 메모리 내 행렬의 첫 번째 요소를 가리키는 256비트 정렬 포인터여야 합니다. ldm는 연속된 행(행 우선 레이아웃의 경우) 또는 열(열 우선 레이아웃의 경우) 사이의 요소 단위 스트라이드를 설명하며, __half 요소 타입의 경우 8의 배수이거나 float 요소 타입의 경우 4의 배수여야 합니다. (즉, 두 경우 모두 16바이트의 배수). 출력 행렬의 레이아웃은 mem_row_major 또는 mem_col_major 중 하나로 지정되어야 합니다. mptr, ldm, layout의 값과 a에 대한 모든 템플릿 매개변수는 워프 내 모든 스레드에서 동일해야 합니다.

fill_fragment

행렬 프래그먼트를 상수 값 v로 채웁니다. 행렬 요소가 각 프래그먼트에 매핑되는 방식은 미지정(unspecified)이므로, 이 함수는 일반적으로 워프 내 모든 스레드가 v에 대해 공통 값을 사용하여 호출합니다.

mma_sync

모든 워프 레인이 mma_sync에 도달할 때까지 대기한 다음, 워프 동기 행렬 곱셈-누산(multiply-accumulate) 연산 D=A*B+C을 수행합니다. 인플레이스(in-place) 연산인 C=A*B+C도 지원됩니다. satf의 값과 각 행렬 프래그먼트에 대한 템플릿 매개변수는 워프 내 모든 스레드에서 동일해야 합니다. 또한 템플릿 매개변수 m, n 및 k는 프래그먼트 A, B, C 및 D 간에 일치해야 합니다. 이 함수는 워프 내 모든 스레드에서 호출되어야 하며, 그렇지 않으면 결과는 정의되지 않습니다.

satf (유한 값으로 포화(saturate)) 모드가 true이면, 목적지 누산기에 대해 다음과 같은 추가 수치적 속성이 적용됩니다:

  • 요소 결과가 +Infinity이면, 해당 누산기는 +MAX_NORM를 포함합니다

  • 요소 결과가 -Infinity이면, 해당 누산기는 -MAX_NORM를 포함합니다

  • 요소 결과가 NaN이면, 해당 누산기는 +0를 포함합니다

행렬 요소가 각 스레드의 fragment에 매핑되는 방식은 미지정(unspecified)이므로, 개별 행렬 요소는 store_matrix_sync을 호출한 후 메모리(공유 또는 전역)에서 액세스해야 합니다. 워프 내 모든 스레드가 모든 프래그먼트 요소에 대해 요소별(element-wise) 연산을 균일하게 적용하는 특수한 경우에는, 다음의 fragment 클래스 멤버를 사용하여 직접 요소 액세스를 구현할 수 있습니다.

enum fragment<Use, m, n, k, T, Layout>::num_elements;
T fragment<Use, m, n, k, T, Layout>::x[num_elements];

예를 들어, 다음 코드는 accumulator 행렬 타일을 절반으로 스케일합니다.

wmma::fragment<wmma::accumulator, 16, 16, 16, float> frag;
float alpha = 0.5f; // Same value for all threads in warp
/*...*/
for(int t=0; t<frag.num_elements; t++)
frag.x[t] *= alpha;

5.4.11.2. 대체 부동소수점#

텐서 코어(Tensor Cores)는 컴퓨트 기능(compute capability) 8.0 이상인 디바이스에서 대체 유형의 부동소수점 연산을 지원합니다.

__nv_bfloat16

이 데이터 형식은 f32와 동일한 범위를 갖지만 정밀도가 감소한(7비트) 대체 fp16 형식입니다. 이 데이터 형식은 다음과 함께 직접 사용할 수 있습니다 __nv_bfloat16 사용 가능한 타입 cuda_bf16.h. 다음을 갖는 행렬 프래그먼트 __nv_bfloat16 데이터 타입은 다음의 누산기와 함께 조합되어야 합니다 float 타입. 지원되는 형상과 연산은 다음과 동일합니다 __half.

tf32

이 데이터 형식은 텐서 코어(Tensor Cores)가 지원하는 특수 부동소수점 형식으로, f32와 동일한 범위를 가지며 정밀도가 감소합니다(>=10비트). 이 형식의 내부 레이아웃은 구현 정의(implementation-defined)입니다. 이 부동소수점 형식을 WMMA 연산과 함께 사용하려면, 입력 행렬을 tf32 정밀도로 수동 변환해야 합니다.

변환을 용이하게 하기 위해, 새로운 내장 함수(intrinsic) __float_to_tf32가 제공됩니다. 이 내장 함수의 입력 및 출력 인수는 float 타입이지만, 출력은 수치적으로 tf32가 됩니다. 이 새로운 정밀도는 텐서 코어(Tensor Cores)에서만 사용하도록 의도되었으며, 다른 floattype 연산과 혼용할 경우 결과의 정밀도와 범위는 정의되지 않습니다.

입력 행렬( matrix_a 또는 matrix_b)이 tf32 정밀도로 변환되면, a의 조합 fragment 와 함께 precision::tf32 정밀도, 그리고 데이터 타입이 float 로 load_matrix_sync 이 새로운 기능을 활용할 것입니다. 두 누산기 프래그먼트는 모두 float 데이터 타입입니다. 지원되는 유일한 행렬 크기는 16x16x8(m-n-k)입니다.

프래그먼트의 요소는 다음과 같이 표현됩니다 float, 따라서 다음으로부터의 매핑은 element_type<T> 로 storage_element_type<T> 입니다:

precision::tf32 -> float

5.4.11.3. 배정밀도#

텐서 코어(Tensor Cores)는 컴퓨트 기능(compute capability) 8.0 이상인 디바이스에서 배정밀도 부동소수점 연산을 지원합니다. 이 새로운 기능을 사용하려면, a fragment 와 함께 double 타입을 사용해야 합니다. 이 mma_sync 연산은 .rn(가장 가까운 짝수로 반올림) 반올림 수정자와 함께 수행됩니다.

5.4.11.4. 서브바이트(sub-byte) 연산#

서브바이트(sub-byte) WMMA 연산은 텐서 코어(Tensor Cores)의 저정밀 기능에 액세스하는 방법을 제공합니다. 이는 프리뷰 기능으로 간주되며, 즉 이를 위한 데이터 구조와 API는 변경될 수 있고 향후 릴리스와 호환되지 않을 수 있습니다. 이 기능은 nvcuda::wmma::experimental 네임스페이스를 통해 사용할 수 있습니다:

namespace experimental {
    namespace precision {
        struct u4; // 4-bit unsigned
        struct s4; // 4-bit signed
        struct b1; // 1-bit
   }
    enum bmmaBitOp {
        bmmaBitOpXOR = 1, // compute_75 minimum
        bmmaBitOpAND = 2  // compute_80 minimum
    };
    enum bmmaAccumulateOp { bmmaAccumulateOpPOPC = 1 };
}

4비트 정밀도의 경우 사용 가능한 API는 동일하게 유지되지만, 프래그먼트 데이터 타입으로 experimental::precision::u4 또는 experimental::precision::s4을 지정해야 합니다. 프래그먼트의 요소가 함께 패킹되어 있으므로, 해당 프래그먼트에서 num_storage_elements는 num_elements보다 더 작습니다. 따라서 서브바이트 프래그먼트의 num_elements 변수는 서브바이트 타입 element_type<T> 요소의 개수를 반환합니다. 이는 1비트 정밀도에서도 마찬가지이며, 이 경우 element_type<T>에서 storage_element_type<T>으로의 매핑은 다음과 같습니다:

experimental::precision::u4 -> unsigned (8 elements in 1 storage element)
experimental::precision::s4 -> int (8 elements in 1 storage element)
experimental::precision::b1 -> unsigned (32 elements in 1 storage element)
T -> T  //all other types

서브바이트(sub-byte) 프래그먼트에 대해 허용되는 레이아웃은 항상 row_major 에 대해 matrix_a 그리고 col_major 에 대해 matrix_b.

서브바이트(sub-byte) 연산의 경우 값은 ldm 에서 load_matrix_sync 요소 타입에 대해 32의 배수여야 합니다 experimental::precision::u4 그리고 experimental::precision::s4 또는 요소 타입에 대해 128의 배수 experimental::precision::b1 (즉, 두 경우 모두 16바이트의 배수).

참고

MMA 명령에 대해 다음 변형(variant)에 대한 지원은 더 이상 권장되지 않으며(deprecated) sm_90에서 제거될 것입니다:

  • experimental::precision::u4

  • experimental::precision::s4

  • experimental::precision::b1 와 함께 bmmaBitOp 설정됨 bmmaBitOpXOR

bmma_sync

모든 워프 레인(warp lanes)이 bmma_sync를 실행할 때까지 대기한 다음, 워프 동기식 비트 행렬 곱셈-누산(multiply-accumulate) 연산 D = (A op B) + C을 수행합니다. 여기서 op(은)는 논리 연산 bmmaBitOp 다음에 bmmaAccumulateOp에 의해 정의된 누산을 수행하는 것으로 구성됩니다. 사용 가능한 연산은 다음과 같습니다:

bmmaBitOpXOR, matrix_a의 한 행과 matrix_b의 128비트 열에 대한 128비트 XOR

bmmaBitOpAND, matrix_a의 한 행과 matrix_b의 128비트 열에 대한 128비트 AND, 컴퓨트 기능(compute capability) 8.0 이상인 디바이스에서 사용 가능합니다.

누산 op는 항상 bmmaAccumulateOpPOPC이며, 설정된 비트의 개수를 셉니다.

5.4.11.5. 제한 사항#

텐서 코어(Tensor Cores)에 필요한 특수 형식은 각 주요 및 부(major/minor) 디바이스 아키텍처마다 다를 수 있습니다. 이는 스레드가 전체 행렬의 프래그먼트(불투명한 아키텍처별 ABI 데이터 구조)만을 보유하며, 개발자가 개별 매개변수가 행렬 곱셈-누산(multiply-accumulate)에 참여하는 레지스터에 어떻게 매핑되는지에 대해 가정할 수 없다는 점 때문에 더욱 복잡해집니다.

프래그먼트는 아키텍처별이므로, 서로 다른 링크 호환 아키텍처에 대해 컴파일되고 동일한 디바이스 실행 파일로 함께 링크된 함수라면, 함수 A에서 함수 B로 이를 전달하는 것은 안전하지 않습니다. 이 경우 프래그먼트의 크기와 레이아웃은 한 아키텍처에만 특정되며, 다른 쪽에서 WMMA API를 사용하면 잘못된 결과 또는 잠재적으로 손상(corruption)을 초래합니다.

프래그먼트의 레이아웃이 다른 링크 호환 아키텍처의 예로는 sm_70과 sm_75가 있습니다.

fragA.cu: void foo() { wmma::fragment<...> mat_a; bar(&mat_a); }
fragB.cu: void bar(wmma::fragment<...> *mat_a) { // operate on mat_a }

// sm_70 fragment layout
$> nvcc -dc -arch=compute_70 -code=sm_70 fragA.cu -o fragA.o
// sm_75 fragment layout
$> nvcc -dc -arch=compute_75 -code=sm_75 fragB.cu -o fragB.o
// Linking the two together
$> nvcc -dlink -arch=sm_75 fragA.o fragB.o -o frag.o

이러한 미지정(unspecified) 동작은 컴파일 시점이나 런타임의 도구에서도 감지되지 않을 수 있으므로, 프래그먼트의 레이아웃이 일관되도록 보장하기 위해 각별한 주의가 필요합니다. 이 링크 위험은 다른 링크 호환 아키텍처를 대상으로 빌드되었고 WMMA 프래그먼트를 전달받을 것으로 기대하는 레거시 라이브러리와 링크할 때 가장 발생하기 쉽습니다.

약한 링크(예: CUDA C++ 인라인 함수)의 경우, 링커는 사용 가능한 어떤 함수 정의든 선택할 수 있으며, 그 결과 컴파일 유닛 간에 암묵적인 전달이 발생할 수 있다는 점에 유의하십시오.

이러한 종류의 문제를 피하려면, 행렬은 외부 인터페이스(예: wmma::store_matrix_sync(dst, …);)를 통해 전달되기 위해 항상 메모리에 저장되어야 하며, 그 다음 포인터 타입으로 bar()에 안전하게 전달할 수 있습니다 [예: float *dst].

sm_70은 sm_75에서 실행될 수 있으므로, 위의 예시 sm_75 코드는 sm_70으로 변경해도 sm_75에서 올바르게 동작할 수 있습니다. 그러나 다른 sm_75 분리 컴파일 바이너리와 링크할 때는 애플리케이션에 sm_75 네이티브 코드를 포함하는 것을 권장합니다.

5.4.11.7. 예제#

다음 코드는 단일 워프에서 16x16x16 행렬 곱셈을 구현합니다.

#include <mma.h>
using namespace nvcuda;

__global__ void wmma_ker(half *a, half *b, float *c) {
   // Declare the fragments
   wmma::fragment<wmma::matrix_a, 16, 16, 16, half, wmma::col_major> a_frag;
   wmma::fragment<wmma::matrix_b, 16, 16, 16, half, wmma::row_major> b_frag;
   wmma::fragment<wmma::accumulator, 16, 16, 16, float> c_frag;

   // Initialize the output to zero
   wmma::fill_fragment(c_frag, 0.0f);

   // Load the inputs
   wmma::load_matrix_sync(a_frag, a, 16);
   wmma::load_matrix_sync(b_frag, b, 16);

   // Perform the matrix multiplication
   wmma::mma_sync(c_frag, a_frag, b_frag, c_frag);

   // Store the output
   wmma::store_matrix_sync(c, c_frag, 16, wmma::mem_row_major);
}
